## Step 2: Complete `lib.rs`

```rust
//! Roman numeral converter using subtractive notation.
//!
//! Supports values in `1..=3999`. Only the canonical subtractive form
//! is accepted on input; sequences like `IIII`, `VV`, `IC`, or lowercase
//! letters are rejected.

use std::fmt;

/// Errors returned by the Roman numeral conversion functions.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    /// `to_roman` was called with `0` or a value greater than `3999`.
    OutOfRange(u32),
    /// `from_roman` was called with an empty string.
    Empty,
    /// `from_roman` encountered a character that is not one of `IVXLCDM`.
    InvalidCharacter(char),
    /// `from_roman` parsed something that is not the canonical
    /// subtractive form for any `n` in `1..=3999`.
    NotCanonical,
}

impl fmt::Display for RomanError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            RomanError::OutOfRange(n) => {
                write!(f, "value {} is out of range (must be 1..=3999)", n)
            }
            RomanError::Empty => write!(f, "empty input string"),
            RomanError::InvalidCharacter(c) => {
                write!(f, "invalid character {:?} (allowed: I V X L C D M)", c)
            }
            RomanError::NotCanonical => {
                write!(f, "input is not the canonical Roman numeral form")
            }
        }
    }
}

/// Ordered (largest -> smallest) table of value/symbol pairs used for
/// greedy encoding. Includes the six subtractive combinations.
const PAIRS: &[(u32, &str)] = &[
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
];

/// Value of a single Roman digit, or `None` if the character is not
/// one of the seven allowed symbols.
fn char_value(c: char) -> Option<u32> {
    match c {
        'I' => Some(1),
        'V' => Some(5),
        'X' => Some(10),
        'L' => Some(50),
        'C' => Some(100),
        'D' => Some(500),
        'M' => Some(1000),
        _ => None,
    }
}

/// Convert an integer in `1..=3999` to its canonical Roman numeral string.
pub fn to_roman(n: u32) -> Result<String, RomanError> {
    if n == 0 || n > 3999 {
        return Err(RomanError::OutOfRange(n));
    }
    let mut remaining = n;
    let mut out = String::new();
    for &(value, symbol) in PAIRS {
        while remaining >= value {
            out.push_str(symbol);
            remaining -= value;
        }
    }
    Ok(out)
}

/// Parse a canonical Roman numeral string back to its integer value.
pub fn from_roman(s: &str) -> Result<u32, RomanError> {
    if s.is_empty() {
        return Err(RomanError::Empty);
    }

    // Validate character set first (also rejects lowercase).
    let chars: Vec<char> = s.chars().collect();
    for &c in &chars {
        if char_value(c).is_none() {
            return Err(RomanError::InvalidCharacter(c));
        }
    }

    // Parse using the standard subtractive walk. We accumulate in i64
    // so that pathological long inputs cannot wrap a u32.
    let mut total: i64 = 0;
    for i in 0..chars.len() {
        // Safe: every char was validated above.
        let v = char_value(chars[i]).unwrap() as i64;
        let next = chars.get(i + 1).and_then(|&c| char_value(c)).unwrap_or(0) as i64;
        if v < next {
            total -= v;
        } else {
            total += v;
        }
    }

    if total < 1 || total > 3999 {
        return Err(RomanError::NotCanonical);
    }
    let n = total as u32;

    // Canonical-form check: re-encode and compare. This rejects
    // things like "IIII", "VV", "IC", "LL", "IIX", etc.
    let canonical = to_roman(n).map_err(|_| RomanError::NotCanonical)?;
    if canonical != s {
        return Err(RomanError::NotCanonical);
    }

    Ok(n)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn to_roman_basic() {
        assert_eq!(to_roman(1).unwrap(), "I");
        assert_eq!(to_roman(4).unwrap(), "IV");
        assert_eq!(to_roman(9).unwrap(), "IX");
        assert_eq!(to_roman(40).unwrap(), "XL");
        assert_eq!(to_roman(90).unwrap(), "XC");
        assert_eq!(to_roman(400).unwrap(), "CD");
        assert_eq!(to_roman(900).unwrap(), "CM");
        assert_eq!(to_roman(1994).unwrap(), "MCMXCIV");
        assert_eq!(to_roman(3999).unwrap(), "MMMCMXCIX");
    }

    #[test]
    fn to_roman_out_of_range() {
        assert_eq!(to_roman(0), Err(RomanError::OutOfRange(0)));
        assert_eq!(to_roman(4000), Err(RomanError::OutOfRange(4000)));
    }

    #[test]
    fn from_roman_basic() {
        assert_eq!(from_roman("I").unwrap(), 1);
        assert_eq!(from_roman("IV").unwrap(), 4);
        assert_eq!(from_roman("MCMXCIV").unwrap(), 1994);
        assert_eq!(from_roman("MMMCMXCIX").unwrap(), 3999);
    }

    #[test]
    fn from_roman_errors() {
        assert_eq!(from_roman(""), Err(RomanError::Empty));
        assert_eq!(from_roman("iv"), Err(RomanError::InvalidCharacter('i')));
        assert_eq!(from_roman("IIX"), Err(RomanError::NotCanonical));
        assert_eq!(from_roman("IIII"), Err(RomanError::NotCanonical));
        assert_eq!(from_roman("VV"), Err(RomanError::NotCanonical));
        assert_eq!(from_roman("IC"), Err(RomanError::NotCanonical));
        assert_eq!(from_roman("MMMM"), Err(RomanError::NotCanonical));
    }

    #[test]
    fn round_trip_full_range() {
        for n in 1u32..=3999 {
            let s = to_roman(n).unwrap();
            assert_eq!(from_roman(&s), Ok(n), "round trip failed for {}", n);
        }
    }

    #[test]
    fn display_messages() {
        assert!(format!("{}", RomanError::OutOfRange(0)).contains("0"));
        assert!(format!("{}", RomanError::Empty).contains("empty"));
        assert!(format!("{}", RomanError::InvalidCharacter('z')).contains('z'));
        assert!(!format!("{}", RomanError::NotCanonical).is_empty());
    }
}
```
