//! Levenshtein edit distance over Unicode code points.
//!
//! Computes the minimum number of single-character insertions, deletions, or
//! substitutions required to transform one string into another. The distance is
//! measured in `char`s (Unicode scalar values), not bytes, so multi-byte
//! characters count as a single edit.

/// Returns the Levenshtein distance between `a` and `b`.
///
/// The distance is the minimum number of single-character insertions,
/// deletions, or substitutions required to transform `a` into `b`. Operates
/// on `char`s, so `distance("é", "e") == 1` regardless of UTF-8 byte length.
///
/// # Examples
///
/// ```
/// use levenshtein::distance;
///
/// assert_eq!(distance("", ""), 0);
/// assert_eq!(distance("kitten", "sitting"), 3);
/// assert_eq!(distance("é", "e"), 1);
/// ```
pub fn distance(a: &str, b: &str) -> usize {
    // Fast paths for empty inputs avoid allocating.
    if a.is_empty() {
        return b.chars().count();
    }
    if b.is_empty() {
        return a.chars().count();
    }

    // Collect once so we can index by position and know the length up front.
    // Using the shorter string as the "inner" axis keeps the row buffers small.
    let a_chars: Vec<char> = a.chars().collect();
    let b_chars: Vec<char> = b.chars().collect();

    let (short, long) = if a_chars.len() <= b_chars.len() {
        (&a_chars, &b_chars)
    } else {
        (&b_chars, &a_chars)
    };

    let m = short.len();
    let n = long.len();

    // Two-row dynamic programming. `prev[j]` is the distance between
    // `long[..i-1]` and `short[..j]`; `curr[j]` is the distance for
    // `long[..i]` and `short[..j]`.
    let mut prev: Vec<usize> = (0..=m).collect();
    let mut curr: Vec<usize> = vec![0; m + 1];

    for i in 1..=n {
        curr[0] = i;
        let long_char = long[i - 1];

        for j in 1..=m {
            let cost = if long_char == short[j - 1] { 0 } else { 1 };

            let deletion = prev[j] + 1;          // drop a char from `long`
            let insertion = curr[j - 1] + 1;     // insert into `long`
            let substitution = prev[j - 1] + cost;

            curr[j] = deletion.min(insertion).min(substitution);
        }

        std::mem::swap(&mut prev, &mut curr);
    }

    prev[m]
}

#[cfg(test)]
mod tests {
    use super::distance;

    #[test]
    fn identity() {
        assert_eq!(distance("", ""), 0);
        assert_eq!(distance("a", "a"), 0);
        assert_eq!(distance("hello", "hello"), 0);
        assert_eq!(distance("naïveté", "naïveté"), 0);
    }

    #[test]
    fn empty_baseline() {
        assert_eq!(distance("", "abc"), 3);
        assert_eq!(distance("abc", ""), 3);
        assert_eq!(distance("", "é"), 1);
        assert_eq!(distance("", "naïveté"), "naïveté".chars().count());
    }

    #[test]
    fn symmetry() {
        let pairs = [
            ("kitten", "sitting"),
            ("flaw", "lawn"),
            ("gumbo", "gambol"),
            ("é", "e"),
            ("abc", ""),
            ("rosettacode", "raisethysword"),
        ];
        for (a, b) in pairs {
            assert_eq!(distance(a, b), distance(b, a), "asymmetry on ({a:?}, {b:?})");
        }
    }

    #[test]
    fn classic_examples() {
        assert_eq!(distance("kitten", "sitting"), 3);
        assert_eq!(distance("flaw", "lawn"), 2);
        assert_eq!(distance("gumbo", "gambol"), 2);
        assert_eq!(distance("book", "back"), 2);
    }

    #[test]
    fn unicode_code_points() {
        // "é" is two bytes in UTF-8 but one char.
        assert_eq!(distance("é", "e"), 1);
        assert_eq!("é".len(), 2);
        assert_eq!("é".chars().count(), 1);

        // Emoji are single chars too (when not in a ZWJ sequence).
        assert_eq!(distance("a🎉b", "ab"), 1);
        assert_eq!(distance("🎉", ""), 1);
        assert_eq!(distance("🎉", "🎊"), 1);
    }

    #[test]
    fn single_edit_kinds() {
        assert_eq!(distance("abc", "abcd"), 1); // insertion
        assert_eq!(distance("abcd", "abc"), 1); // deletion
        assert_eq!(distance("abc", "abd"), 1); // substitution
    }
}
