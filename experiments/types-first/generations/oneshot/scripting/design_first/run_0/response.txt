# Step 2 — Complete `lib.rs`

```rust
//! Tiny expression-language evaluator.
//!
//! Grammar:
//!   expr  := let | if | or
//!   let   := "let" IDENT "=" expr "in" expr
//!   if    := "if" expr "then" expr "else" expr
//!   or    := and ("or" and)*
//!   and   := not ("and" not)*
//!   not   := "not" not | cmp
//!   cmp   := add (("=="|"!="|"<"|"<="|">"|">=") add)?
//!   add   := mul (("+"|"-") mul)*
//!   mul   := unary (("*"|"/") unary)*
//!   unary := "-" unary | atom
//!   atom  := INT | "true" | "false" | IDENT | "(" expr ")"

use std::fmt;

// ====================================================================
// AST
// ====================================================================

#[derive(Debug, Clone)]
pub enum Expr {
    Int(i64),
    Bool(bool),
    Ident(String),
    Let(String, Box<Expr>, Box<Expr>),
    If(Box<Expr>, Box<Expr>, Box<Expr>),
    BinOp(BinOp, Box<Expr>, Box<Expr>),
    UnOp(UnOp, Box<Expr>),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum BinOp {
    Or, And,
    Eq, Ne, Lt, Le, Gt, Ge,
    Add, Sub, Mul, Div,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum UnOp {
    Neg, Not,
}

// ====================================================================
// Values
// ====================================================================

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Value {
    Int(i64),
    Bool(bool),
}

impl fmt::Display for Value {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Value::Int(n) => write!(f, "{}", n),
            Value::Bool(b) => write!(f, "{}", b),
        }
    }
}

// ====================================================================
// Errors
// ====================================================================

#[derive(Debug, Clone)]
pub struct ParseError {
    pub offset: usize,
    pub message: String,
}

impl fmt::Display for ParseError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "parse error at byte offset {}: {}", self.offset, self.message)
    }
}

#[derive(Debug, Clone)]
pub enum EvalError {
    TypeError(String),
    DivisionByZero,
    UnboundIdentifier(String),
}

impl fmt::Display for EvalError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            EvalError::TypeError(msg) => write!(f, "type error: {}", msg),
            EvalError::DivisionByZero => write!(f, "division by zero"),
            EvalError::UnboundIdentifier(name) => write!(f, "unbound identifier: {}", name),
        }
    }
}

// ====================================================================
// Lexer
// ====================================================================

#[derive(Debug, Clone)]
enum TokenKind {
    Int(i64),
    Ident(String),
    // keywords
    Let, In, If, Then, Else, Or, And, Not, True, False,
    // operators / punctuation
    EqEq, NotEq, Lt, Le, Gt, Ge,
    Plus, Minus, Star, Slash,
    Assign,
    LParen, RParen,
    Eof,
}

#[derive(Debug, Clone)]
struct Token {
    kind: TokenKind,
    offset: usize,
}

fn tokenize(input: &str) -> Result<Vec<Token>, ParseError> {
    let bytes = input.as_bytes();
    let mut i = 0usize;
    let mut tokens: Vec<Token> = Vec::new();

    while i < bytes.len() {
        // skip whitespace
        while i < bytes.len() && bytes[i].is_ascii_whitespace() {
            i += 1;
        }
        if i >= bytes.len() {
            break;
        }
        let start = i;
        let b = bytes[i];

        if b.is_ascii_digit() {
            let mut n: i64 = 0;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                let d = (bytes[i] - b'0') as i64;
                n = n
                    .checked_mul(10)
                    .and_then(|x| x.checked_add(d))
                    .ok_or_else(|| ParseError {
                        offset: start,
                        message: "integer literal out of range".to_string(),
                    })?;
                i += 1;
            }
            tokens.push(Token { kind: TokenKind::Int(n), offset: start });
        } else if b == b'_' || (b'a'..=b'z').contains(&b) {
            while i < bytes.len()
                && (bytes[i] == b'_'
                    || (b'a'..=b'z').contains(&bytes[i])
                    || bytes[i].is_ascii_digit())
            {
                i += 1;
            }
            let s = &input[start..i];
            let kind = match s {
                "let" => TokenKind::Let,
                "in" => TokenKind::In,
                "if" => TokenKind::If,
                "then" => TokenKind::Then,
                "else" => TokenKind::Else,
                "or" => TokenKind::Or,
                "and" => TokenKind::And,
                "not" => TokenKind::Not,
                "true" => TokenKind::True,
                "false" => TokenKind::False,
                other => TokenKind::Ident(other.to_string()),
            };
            tokens.push(Token { kind, offset: start });
        } else {
            let (kind, len) = match b {
                b'+' => (TokenKind::Plus, 1),
                b'-' => (TokenKind::Minus, 1),
                b'*' => (TokenKind::Star, 1),
                b'/' => (TokenKind::Slash, 1),
                b'(' => (TokenKind::LParen, 1),
                b')' => (TokenKind::RParen, 1),
                b'=' => {
                    if i + 1 < bytes.len() && bytes[i + 1] == b'=' {
                        (TokenKind::EqEq, 2)
                    } else {
                        (TokenKind::Assign, 1)
                    }
                }
                b'!' => {
                    if i + 1 < bytes.len() && bytes[i + 1] == b'=' {
                        (TokenKind::NotEq, 2)
                    } else {
                        return Err(ParseError {
                            offset: i,
                            message: "unexpected character '!'".to_string(),
                        });
                    }
                }
                b'<' => {
                    if i + 1 < bytes.len() && bytes[i + 1] == b'=' {
                        (TokenKind::Le, 2)
                    } else {
                        (TokenKind::Lt, 1)
                    }
                }
                b'>' => {
                    if i + 1 < bytes.len() && bytes[i + 1] == b'=' {
                        (TokenKind::Ge, 2)
                    } else {
                        (TokenKind::Gt, 1)
                    }
                }
                _ => {
                    return Err(ParseError {
                        offset: i,
                        message: format!("unexpected character {:?}", b as char),
                    });
                }
            };
            tokens.push(Token { kind, offset: start });
            i += len;
        }
    }

    tokens.push(Token { kind: TokenKind::Eof, offset: input.len() });
    Ok(tokens)
}

// ====================================================================
// Parser
// ====================================================================

struct Parser {
    tokens: Vec<Token>,
    pos: usize,
}

impl Parser {
    fn peek(&self) -> &Token {
        &self.tokens[self.pos]
    }

    fn bump(&mut self) -> Token {
        let t = self.tokens[self.pos].clone();
        self.pos += 1;
        t
    }

    fn err(&self, msg: impl Into<String>) -> ParseError {
        ParseError { offset: self.peek().offset, message: msg.into() }
    }

    fn parse_program(&mut self) -> Result<Expr, ParseError> {
        let e = self.parse_expr()?;
        match self.peek().kind {
            TokenKind::Eof => Ok(e),
            _ => Err(self.err("expected end of input")),
        }
    }

    fn parse_expr(&mut self) -> Result<Expr, ParseError> {
        match self.peek().kind {
            TokenKind::Let => self.parse_let(),
            TokenKind::If => self.parse_if(),
            _ => self.parse_or(),
        }
    }

    fn parse_let(&mut self) -> Result<Expr, ParseError> {
        // 'let'
        self.bump();
        let name = match self.peek().kind {
            TokenKind::Ident(_) => {
                if let TokenKind::Ident(s) = self.bump().kind { s } else { unreachable!() }
            }
            _ => return Err(self.err("expected identifier after 'let'")),
        };
        match self.peek().kind {
            TokenKind::Assign => { self.bump(); }
            _ => return Err(self.err("expected '=' in let")),
        }
        let value = self.parse_expr()?;
        match self.peek().kind {
            TokenKind::In => { self.bump(); }
            _ => return Err(self.err("expected 'in' in let")),
        }
        let body = self.parse_expr()?;
        Ok(Expr::Let(name, Box::new(value), Box::new(body)))
    }

    fn parse_if(&mut self) -> Result<Expr, ParseError> {
        // 'if'
        self.bump();
        let cond = self.parse_expr()?;
        match self.peek().kind {
            TokenKind::Then => { self.bump(); }
            _ => return Err(self.err("expected 'then' in if")),
        }
        let t = self.parse_expr()?;
        match self.peek().kind {
            TokenKind::Else => { self.bump(); }
            _ => return Err(self.err("expected 'else' in if")),
        }
        let e = self.parse_expr()?;
        Ok(Expr::If(Box::new(cond), Box::new(t), Box::new(e)))
    }

    fn parse_or(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_and()?;
        while matches!(self.peek().kind, TokenKind::Or) {
            self.bump();
            let right = self.parse_and()?;
            left = Expr::BinOp(BinOp::Or, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_and(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_not()?;
        while matches!(self.peek().kind, TokenKind::And) {
            self.bump();
            let right = self.parse_not()?;
            left = Expr::BinOp(BinOp::And, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_not(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek().kind, TokenKind::Not) {
            self.bump();
            let inner = self.parse_not()?;
            Ok(Expr::UnOp(UnOp::Not, Box::new(inner)))
        } else {
            self.parse_cmp()
        }
    }

    fn parse_cmp(&mut self) -> Result<Expr, ParseError> {
        let left = self.parse_add()?;
        let op = match self.peek().kind {
            TokenKind::EqEq => BinOp::Eq,
            TokenKind::NotEq => BinOp::Ne,
            TokenKind::Lt => BinOp::Lt,
            TokenKind::Le => BinOp::Le,
            TokenKind::Gt => BinOp::Gt,
            TokenKind::Ge => BinOp::Ge,
            _ => return Ok(left),
        };
        self.bump();
        let right = self.parse_add()?;
        Ok(Expr::BinOp(op, Box::new(left), Box::new(right)))
    }

    fn parse_add(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_mul()?;
        loop {
            let op = match self.peek().kind {
                TokenKind::Plus => BinOp::Add,
                TokenKind::Minus => BinOp::Sub,
                _ => break,
            };
            self.bump();
            let right = self.parse_mul()?;
            left = Expr::BinOp(op, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_mul(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_unary()?;
        loop {
            let op = match self.peek().kind {
                TokenKind::Star => BinOp::Mul,
                TokenKind::Slash => BinOp::Div,
                _ => break,
            };
            self.bump();
            let right = self.parse_unary()?;
            left = Expr::BinOp(op, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_unary(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek().kind, TokenKind::Minus) {
            self.bump();
            let inner = self.parse_unary()?;
            Ok(Expr::UnOp(UnOp::Neg, Box::new(inner)))
        } else {
            self.parse_atom()
        }
    }

    fn parse_atom(&mut self) -> Result<Expr, ParseError> {
        let tok = self.peek().clone();
        match tok.kind {
            TokenKind::Int(n) => { self.bump(); Ok(Expr::Int(n)) }
            TokenKind::True => { self.bump(); Ok(Expr::Bool(true)) }
            TokenKind::False => { self.bump(); Ok(Expr::Bool(false)) }
            TokenKind::Ident(s) => { self.bump(); Ok(Expr::Ident(s)) }
            TokenKind::LParen => {
                self.bump();
                let inner = self.parse_expr()?;
                match self.peek().kind {
                    TokenKind::RParen => { self.bump(); Ok(inner) }
                    _ => Err(self.err("expected ')'")),
                }
            }
            TokenKind::Eof => Err(ParseError {
                offset: tok.offset,
                message: "unexpected end of input".to_string(),
            }),
            _ => Err(ParseError {
                offset: tok.offset,
                message: "expected an atom (integer, boolean, identifier, or '(')".to_string(),
            }),
        }
    }
}

pub fn parse(input: &str) -> Result<Expr, ParseError> {
    let tokens = tokenize(input)?;
    let mut p = Parser { tokens, pos: 0 };
    p.parse_program()
}

// Evaluator

pub fn eval(expr: &Expr) -> Result<Value, EvalError> {
    let mut env: Vec<(String, Value)> = Vec::new();
    eval_with(expr, &mut env)
}

fn lookup(env: &[(String, Value)], name: &str) -> Option<Value> {
    for (n, v) in env.iter().rev() {
        if n == name {
            return Some(*v);
        }
    }
    None
}

fn eval_with(expr: &Expr, env: &mut Vec<(String, Value)>) -> Result<Value, EvalError> {
    match expr {
        Expr::Int(n) => Ok(Value::Int(*n)),
        Expr::Bool(b) => Ok(Value::Bool(*b)),
        Expr::Ident(name) => lookup(env, name)
            .ok_or_else(|| EvalError::UnboundIdentifier(name.clone())),
        Expr::Let(name, value, body) => {
            let v = eval_with(value, env)?;
            env.push((name.clone(), v));
            let result = eval_with(body, env);
            env.pop();
            result
        }
        Expr::If(cond, then_b, else_b) => {
            let c = eval_with(cond, env)?;
            match c {
                Value::Bool(true) => eval_with(then_b, env),
                Value::Bool(false) => eval_with(else_b, env),
                Value::Int(_) => Err(EvalError::TypeError(
                    "condition of 'if' must be boolean".to_string(),
                )),
            }
        }
        Expr::UnOp(op, inner) => {
            let v = eval_with(inner, env)?;
            match (op, v) {
                (UnOp::Neg, Value::Int(n)) => n
                    .checked_neg()
                    .map(Value::Int)
                    .ok_or_else(|| EvalError::TypeError("integer overflow on negation".to_string())),
                (UnOp::Neg, Value::Bool(_)) => Err(EvalError::TypeError(
                    "unary '-' expects an integer".to_string(),
                )),
                (UnOp::Not, Value::Bool(b)) => Ok(Value::Bool(!b)),
                (UnOp::Not, Value::Int(_)) => Err(EvalError::TypeError(
                    "'not' expects a boolean".to_string(),
                )),
            }
        }
        Expr::BinOp(op, l, r) => match op {
            BinOp::And => {
                let lv = eval_with(l, env)?;
                match lv {
                    Value::Bool(false) => Ok(Value::Bool(false)),
                    Value::Bool(true) => match eval_with(r, env)? {
                        Value::Bool(b) => Ok(Value::Bool(b)),
                        Value::Int(_) => Err(EvalError::TypeError(
                            "'and' expects boolean operands".to_string(),
                        )),
                    },
                    Value::Int(_) => Err(EvalError::TypeError(
                        "'and' expects boolean operands".to_string(),
                    )),
                }
            }
            BinOp::Or => {
                let lv = eval_with(l, env)?;
                match lv {
                    Value::Bool(true) => Ok(Value::Bool(true)),
                    Value::Bool(false) => match eval_with(r, env)? {
                        Value::Bool(b) => Ok(Value::Bool(b)),
                        Value::Int(_) => Err(EvalError::TypeError(
                            "'or' expects boolean operands".to_string(),
                        )),
                    },
                    Value::Int(_) => Err(EvalError::TypeError(
                        "'or' expects boolean operands".to_string(),
                    )),
                }
            }
            _ => {
                let lv = eval_with(l, env)?;
                let rv = eval_with(r, env)?;
                eval_arith_or_cmp(*op, lv, rv)
            }
        },
    }
}

fn eval_arith_or_cmp(op: BinOp, l: Value, r: Value) -> Result<Value, EvalError> {
    use Value::*;
    match op {
        BinOp::Add => match (l, r) {
            (Int(a), Int(b)) => a
                .checked_add(b)
                .map(Int)
                .ok_or_else(|| EvalError::TypeError("integer overflow in '+'".to_string())),
            _ => Err(EvalError::TypeError("'+' expects integer operands".to_string())),
        },
        BinOp::Sub => match (l, r) {
            (Int(a), Int(b)) => a
                .checked_sub(b)
                .map(Int)
                .ok_or_else(|| EvalError::TypeError("integer overflow in '-'".to_string())),
            _ => Err(EvalError::TypeError("'-' expects integer operands".to_string())),
        },
        BinOp::Mul => match (l, r) {
            (Int(a), Int(b)) => a
                .checked_mul(b)
                .map(Int)
                .ok_or_else(|| EvalError::TypeError("integer overflow in '*'".to_string())),
            _ => Err(EvalError::TypeError("'*' expects integer operands".to_string())),
        },
        BinOp::Div => match (l, r) {
            (Int(_), Int(0)) => Err(EvalError::DivisionByZero),
            (Int(a), Int(b)) => a
                .checked_div(b)
                .map(Int)
                .ok_or_else(|| EvalError::TypeError("integer overflow in '/'".to_string())),
            _ => Err(EvalError::TypeError("'/' expects integer operands".to_string())),
        },
        BinOp::Eq => match (l, r) {
            (Int(a), Int(b)) => Ok(Bool(a == b)),
            (Bool(a), Bool(b)) => Ok(Bool(a == b)),
            _ => Err(EvalError::TypeError(
                "'==' expects operands of the same type".to_string(),
            )),
        },
        BinOp::Ne => match (l, r) {
            (Int(a), Int(b)) => Ok(Bool(a != b)),
            (Bool(a), Bool(b)) => Ok(Bool(a != b)),
            _ => Err(EvalError::TypeError(
                "'!=' expects operands of the same type".to_string(),
            )),
        },
        BinOp::Lt => match (l, r) {
            (Int(a), Int(b)) => Ok(Bool(a < b)),
            _ => Err(EvalError::TypeError("'<' expects integer operands".to_string())),
        },
        BinOp::Le => match (l, r) {
            (Int(a), Int(b)) => Ok(Bool(a <= b)),
            _ => Err(EvalError::TypeError("'<=' expects integer operands".to_string())),
        },
        BinOp::Gt => match (l, r) {
            (Int(a), Int(b)) => Ok(Bool(a > b)),
            _ => Err(EvalError::TypeError("'>' expects integer operands".to_string())),
        },
        BinOp::Ge => match (l, r) {
            (Int(a), Int(b)) => Ok(Bool(a >= b)),
            _ => Err(EvalError::TypeError("'>=' expects integer operands".to_string())),
        },
        BinOp::And | BinOp::Or => unreachable!("and/or are handled before this point"),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn run(src: &str) -> Result<Value, String> {
        let e = parse(src).map_err(|e| e.to_string())?;
        eval(&e).map_err(|e| e.to_string())
    }

    #[test]
    fn integer_arithmetic() {
        assert_eq!(run("1 + 2 * 3").unwrap(), Value::Int(7));
        assert_eq!(run("(1 + 2) * 3").unwrap(), Value::Int(9));
        assert_eq!(run("-3 + 5").unwrap(), Value::Int(2));
        assert_eq!(run("--4").unwrap(), Value::Int(4));
    }

    #[test]
    fn booleans_and_comparisons() {
        assert_eq!(run("true and false").unwrap(), Value::Bool(false));
        assert_eq!(run("true or false").unwrap(), Value::Bool(true));
        assert_eq!(run("not true").unwrap(), Value::Bool(false));
        assert_eq!(run("1 < 2").unwrap(), Value::Bool(true));
        assert_eq!(run("1 == 1 and 2 != 3").unwrap(), Value::Bool(true));
    }

    #[test]
    fn let_and_if() {
        assert_eq!(
            run("let x = 10 in let y = x * 2 in x + y").unwrap(),
            Value::Int(30)
        );
        assert_eq!(
            run("if 1 < 2 then 10 else 20").unwrap(),
            Value::Int(10)
        );
        assert_eq!(
            run("let x = 5 in if x == 5 then true else false").unwrap(),
            Value::Bool(true)
        );
    }

    #[test]
    fn shadowing() {
        assert_eq!(
            run("let x = 1 in let x = 2 in x").unwrap(),
            Value::Int(2)
        );
    }

    #[test]
    fn type_errors() {
        assert!(run("1 + true").is_err());
        assert!(run("if 1 then 2 else 3").is_err());
        assert!(run("true < false").is_err());
        assert!(run("1 == true").is_err());
    }

    #[test]
    fn div_by_zero() {
        match parse("10 / 0").map(|e| eval(&e)).unwrap() {
            Err(EvalError::DivisionByZero) => {}
            other => panic!("expected DivisionByZero, got {:?}", other),
        }
    }

    #[test]
    fn unbound_identifier() {
        match parse("x + 1").map(|e| eval(&e)).unwrap() {
            Err(EvalError::UnboundIdentifier(n)) => assert_eq!(n, "x"),
            other => panic!("expected UnboundIdentifier, got {:?}", other),
        }
    }

    #[test]
    fn parse_error_offset() {
        let err = parse("1 + ").unwrap_err();
        assert_eq!(err.offset, 4);

        let err = parse("1 +! 2").unwrap_err();
        assert_eq!(err.offset, 3);
    }

    #[test]
    fn value_display() {
        assert_eq!(format!("{}", Value::Int(42)), "42");
        assert_eq!(format!("{}", Value::Int(-3)), "-3");
        assert_eq!(format!("{}", Value::Bool(true)), "true");
        assert_eq!(format!("{}", Value::Bool(false)), "false");
    }
}
```
