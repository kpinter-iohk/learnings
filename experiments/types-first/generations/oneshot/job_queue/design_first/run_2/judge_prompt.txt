//! In-memory job queue with retry-on-failure.
//!
//! Standard library only.

use std::collections::HashMap;
use std::fmt;
use std::time::{Duration, Instant};

/// Unique identifier for a job, assigned by the queue at enqueue time.
#[derive(Copy, Clone, Eq, PartialEq, Hash, Debug)]
pub struct JobId(u64);

/// Public job state. Variant names are part of the public API.
#[derive(Debug, PartialEq, Eq)]
pub enum JobState {
    Pending,
    Running,
    FailedPendingRetry,
    Succeeded,
    Dead,
}

/// Errors produced by `Queue` operations.
#[derive(Debug)]
pub enum QueueError {
    /// No job with the given id exists in the queue.
    UnknownJob(JobId),
    /// The job exists but is not currently in the `Running` state.
    NotRunning(JobId),
}

impl fmt::Display for QueueError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            QueueError::UnknownJob(id) => write!(f, "unknown job: {:?}", id),
            QueueError::NotRunning(id) => write!(f, "job {:?} is not currently running", id),
        }
    }
}

/// A handle returned by `Queue::checkout`, exposing the id and payload of
/// a job that has been transitioned to `Running`.
pub struct CheckedOut {
    id: JobId,
    payload: String,
}

impl CheckedOut {
    pub fn id(&self) -> JobId {
        self.id
    }

    pub fn payload(&self) -> &str {
        &self.payload
    }
}

/// Internal representation of a job inside the queue.
struct Job {
    payload: String,
    state: JobState,
    attempts: u32,
    retry_at: Option<Instant>,
}

/// In-memory job queue with bounded retries and exponential back-off.
pub struct Queue {
    max_attempts: u32,
    base_delay: Duration,
    next_id: u64,
    jobs: HashMap<JobId, Job>,
    /// Insertion order of job ids; used to pick the "next" runnable job.
    order: Vec<JobId>,
}

impl Queue {
    /// Create a new queue. `max_attempts` must be `>= 1`.
    pub fn new(max_attempts: u32, base_delay: Duration) -> Self {
        assert!(max_attempts >= 1, "max_attempts must be >= 1");
        Queue {
            max_attempts,
            base_delay,
            next_id: 0,
            jobs: HashMap::new(),
            order: Vec::new(),
        }
    }

    /// Enqueue a new job with the given payload. Returns its freshly assigned id.
    pub fn enqueue(&mut self, payload: String) -> JobId {
        let id = JobId(self.next_id);
        self.next_id += 1;
        let job = Job {
            payload,
            state: JobState::Pending,
            attempts: 0,
            retry_at: None,
        };
        self.jobs.insert(id, job);
        self.order.push(id);
        id
    }

    /// Return the next runnable job (a `Pending` job, or a `FailedPendingRetry`
    /// job whose retry time is `<= now`), transitioning it to `Running`.
    /// Returns `None` if no job is runnable.
    pub fn checkout(&mut self, now: Instant) -> Option<CheckedOut> {
        for id in &self.order {
            if let Some(job) = self.jobs.get(id) {
                let runnable = match job.state {
                    JobState::Pending => true,
                    JobState::FailedPendingRetry => match job.retry_at {
                        Some(t) => t <= now,
                        None => true,
                    },
                    _ => false,
                };
                if runnable {
                    let chosen_id = *id;
                    let job = self.jobs.get_mut(&chosen_id).expect("present");
                    job.state = JobState::Running;
                    job.retry_at = None;
                    return Some(CheckedOut {
                        id: chosen_id,
                        payload: job.payload.clone(),
                    });
                }
            }
        }
        None
    }

    /// Mark a running job as succeeded.
    pub fn succeed(&mut self, id: JobId) -> Result<(), QueueError> {
        let job = self.jobs.get_mut(&id).ok_or(QueueError::UnknownJob(id))?;
        if job.state != JobState::Running {
            return Err(QueueError::NotRunning(id));
        }
        job.state = JobState::Succeeded;
        job.retry_at = None;
        Ok(())
    }

    /// Mark a running job as failed. Increments the attempt counter and either
    /// schedules a retry (with exponential back-off) or marks the job dead.
    pub fn fail(&mut self, id: JobId, now: Instant) -> Result<(), QueueError> {
        let job = self.jobs.get_mut(&id).ok_or(QueueError::UnknownJob(id))?;
        if job.state != JobState::Running {
            return Err(QueueError::NotRunning(id));
        }
        job.attempts += 1;
        let new_attempts = job.attempts;
        if new_attempts < self.max_attempts {
            // base_delay * 2^(new_attempts - 1)
            let shift = new_attempts - 1;
            let multiplier: u32 = 1u32.checked_shl(shift).unwrap_or(u32::MAX);
            let delay = job
                .retry_at // unused, just to silence placeholder
                .map(|_| self.base_delay)
                .unwrap_or(self.base_delay)
                .checked_mul(multiplier)
                .unwrap_or(Duration::from_secs(u64::MAX));
            job.state = JobState::FailedPendingRetry;
            job.retry_at = Some(now + delay);
        } else {
            job.state = JobState::Dead;
            job.retry_at = None;
        }
        Ok(())
    }

    /// Look up the current state of a job, or `None` if no such id is known.
    pub fn get_state(&self, id: JobId) -> Option<JobState> {
        self.jobs.get(&id).map(|j| match j.state {
            JobState::Pending => JobState::Pending,
            JobState::Running => JobState::Running,
            JobState::FailedPendingRetry => JobState::FailedPendingRetry,
            JobState::Succeeded => JobState::Succeeded,
            JobState::Dead => JobState::Dead,
        })
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn enqueue_then_checkout_succeed() {
        let mut q = Queue::new(3, Duration::from_secs(1));
        let id = q.enqueue("payload".to_string());
        assert_eq!(q.get_state(id), Some(JobState::Pending));

        let now = Instant::now();
        let co = q.checkout(now).expect("a runnable job");
        assert_eq!(co.id(), id);
        assert_eq!(co.payload(), "payload");
        assert_eq!(q.get_state(id), Some(JobState::Running));

        // Cannot be checked out again while Running.
        assert!(q.checkout(now).is_none());

        q.succeed(id).unwrap();
        assert_eq!(q.get_state(id), Some(JobState::Succeeded));
    }

    #[test]
    fn fail_retries_with_backoff_then_dies() {
        let base = Duration::from_secs(1);
        let mut q = Queue::new(3, base);
        let id = q.enqueue("p".to_string());
        let t0 = Instant::now();

        // Attempt 1.
        let _ = q.checkout(t0).unwrap();
        q.fail(id, t0).unwrap();
        assert_eq!(q.get_state(id), Some(JobState::FailedPendingRetry));

        // Too early to retry.
        assert!(q.checkout(t0).is_none());
        // Eligible at t0 + base_delay * 2^0 = t0 + 1s.
        let t1 = t0 + base;
        let _ = q.checkout(t1).expect("retry eligible");
        assert_eq!(q.get_state(id), Some(JobState::Running));

        // Attempt 2 fails.
        q.fail(id, t1).unwrap();
        assert_eq!(q.get_state(id), Some(JobState::FailedPendingRetry));
        // Eligible at t1 + base_delay * 2^1 = t1 + 2s.
        assert!(q.checkout(t1 + base).is_none());
        let t2 = t1 + 2 * base;
        let _ = q.checkout(t2).expect("retry eligible");

        // Attempt 3 fails -> Dead.
        q.fail(id, t2).unwrap();
        assert_eq!(q.get_state(id), Some(JobState::Dead));
        assert!(q.checkout(t2 + 100 * base).is_none());
    }

    #[test]
    fn succeed_or_fail_when_not_running_errors() {
        let mut q = Queue::new(2, Duration::from_secs(1));
        let id = q.enqueue("x".to_string());
        let now = Instant::now();
        // Pending, not Running.
        assert!(matches!(q.succeed(id), Err(QueueError::NotRunning(_))));
        assert!(matches!(q.fail(id, now), Err(QueueError::NotRunning(_))));
    }

    #[test]
    fn unknown_job_errors() {
        let mut q = Queue::new(2, Duration::from_secs(1));
        let bogus = JobId(9999);
        assert!(matches!(q.succeed(bogus), Err(QueueError::UnknownJob(_))));
        assert!(matches!(
            q.fail(bogus, Instant::now()),
            Err(QueueError::UnknownJob(_))
        ));
        assert_eq!(q.get_state(bogus), None);
    }

    #[test]
    fn checkout_picks_in_insertion_order() {
        let mut q = Queue::new(2, Duration::from_secs(1));
        let a = q.enqueue("a".to_string());
        let b = q.enqueue("b".to_string());
        let now = Instant::now();
        let first = q.checkout(now).unwrap();
        assert_eq!(first.id(), a);
        let second = q.checkout(now).unwrap();
        assert_eq!(second.id(), b);
        assert!(q.checkout(now).is_none());
    }
}
