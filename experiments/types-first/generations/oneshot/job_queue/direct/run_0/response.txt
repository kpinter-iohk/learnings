```rust
use std::collections::HashMap;
use std::fmt;
use std::time::{Duration, Instant};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum JobState {
    Pending,
    Running,
    FailedPendingRetry,
    Succeeded,
    Dead,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub struct JobId(u64);

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueueError {
    UnknownJob(JobId),
    NotRunning(JobId),
}

impl fmt::Display for QueueError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            QueueError::UnknownJob(id) => {
                write!(f, "unknown job id: {:?}", id)
            }
            QueueError::NotRunning(id) => {
                write!(
                    f,
                    "job {:?} is not currently in the Running state",
                    id
                )
            }
        }
    }
}

pub struct CheckedOut {
    id: JobId,
    payload: String,
}

impl CheckedOut {
    pub fn id(&self) -> JobId {
        self.id
    }

    pub fn payload(&self) -> &str {
        &self.payload
    }
}

struct Job {
    payload: String,
    state: JobState,
    attempts: u32,
    retry_at: Option<Instant>,
    enqueued_seq: u64,
}

pub struct Queue {
    max_attempts: u32,
    base_delay: Duration,
    next_id: u64,
    next_seq: u64,
    jobs: HashMap<JobId, Job>,
}

impl Queue {
    pub fn new(max_attempts: u32, base_delay: Duration) -> Self {
        assert!(max_attempts >= 1, "max_attempts must be >= 1");
        Self {
            max_attempts,
            base_delay,
            next_id: 0,
            next_seq: 0,
            jobs: HashMap::new(),
        }
    }

    pub fn enqueue(&mut self, payload: String) -> JobId {
        let id = JobId(self.next_id);
        self.next_id += 1;
        let seq = self.next_seq;
        self.next_seq += 1;
        self.jobs.insert(
            id,
            Job {
                payload,
                state: JobState::Pending,
                attempts: 0,
                retry_at: None,
                enqueued_seq: seq,
            },
        );
        id
    }

    pub fn checkout(&mut self, now: Instant) -> Option<CheckedOut> {
        // Find runnable job with the smallest enqueued_seq (FIFO among ready jobs).
        let mut best: Option<(u64, JobId)> = None;
        for (id, job) in self.jobs.iter() {
            let runnable = match job.state {
                JobState::Pending => true,
                JobState::FailedPendingRetry => {
                    job.retry_at.map_or(true, |t| t <= now)
                }
                _ => false,
            };
            if runnable {
                let candidate = (job.enqueued_seq, *id);
                match best {
                    None => best = Some(candidate),
                    Some((s, _)) if job.enqueued_seq < s => {
                        best = Some(candidate)
                    }
                    _ => {}
                }
            }
        }

        let (_, id) = best?;
        let job = self.jobs.get_mut(&id).expect("job must exist");
        job.state = JobState::Running;
        Some(CheckedOut {
            id,
            payload: job.payload.clone(),
        })
    }

    pub fn succeed(&mut self, id: JobId) -> Result<(), QueueError> {
        let job = self
            .jobs
            .get_mut(&id)
            .ok_or(QueueError::UnknownJob(id))?;
        if job.state != JobState::Running {
            return Err(QueueError::NotRunning(id));
        }
        job.state = JobState::Succeeded;
        job.retry_at = None;
        Ok(())
    }

    pub fn fail(&mut self, id: JobId, now: Instant) -> Result<(), QueueError> {
        let max_attempts = self.max_attempts;
        let base_delay = self.base_delay;
        let job = self
            .jobs
            .get_mut(&id)
            .ok_or(QueueError::UnknownJob(id))?;
        if job.state != JobState::Running {
            return Err(QueueError::NotRunning(id));
        }
        job.attempts = job.attempts.saturating_add(1);
        if job.attempts < max_attempts {
            let exp = job.attempts - 1;
            let multiplier: u32 = if exp >= 32 {
                u32::MAX
            } else {
                1u32 << exp
            };
            let delay = base_delay
                .checked_mul(multiplier)
                .unwrap_or_else(|| Duration::new(u64::MAX, 0));
            let retry_at = now.checked_add(delay).unwrap_or(now);
            job.retry_at = Some(retry_at);
            job.state = JobState::FailedPendingRetry;
        } else {
            job.state = JobState::Dead;
            job.retry_at = None;
        }
        Ok(())
    }

    pub fn get_state(&self, id: JobId) -> Option<JobState> {
        self.jobs.get(&id).map(|j| j.state)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn enqueue_and_checkout_basic() {
        let mut q = Queue::new(3, Duration::from_millis(100));
        let id = q.enqueue("hello".to_string());
        assert_eq!(q.get_state(id), Some(JobState::Pending));

        let now = Instant::now();
        let co = q.checkout(now).expect("should have a job");
        assert_eq!(co.id(), id);
        assert_eq!(co.payload(), "hello");
        assert_eq!(q.get_state(id), Some(JobState::Running));

        // No more runnable jobs.
        assert!(q.checkout(now).is_none());

        q.succeed(id).expect("succeed");
        assert_eq!(q.get_state(id), Some(JobState::Succeeded));
    }

    #[test]
    fn fail_then_retry() {
        let mut q = Queue::new(3, Duration::from_millis(100));
        let id = q.enqueue("p".to_string());
        let t0 = Instant::now();
        let _co = q.checkout(t0).unwrap();
        q.fail(id, t0).unwrap();
        assert_eq!(q.get_state(id), Some(JobState::FailedPendingRetry));
        // Not yet runnable.
        assert!(q.checkout(t0).is_none());
        // After delay (100ms * 2^0 = 100ms) job is runnable.
        let t1 = t0 + Duration::from_millis(100);
        let co = q.checkout(t1).expect("should be runnable");
        assert_eq!(co.id(), id);
    }

    #[test]
    fn dead_after_max_attempts() {
        let mut q = Queue::new(2, Duration::from_millis(10));
        let id = q.enqueue("p".to_string());
        let t0 = Instant::now();
        let _ = q.checkout(t0).unwrap();
        q.fail(id, t0).unwrap();
        assert_eq!(q.get_state(id), Some(JobState::FailedPendingRetry));
        let t1 = t0 + Duration::from_secs(1);
        let _ = q.checkout(t1).unwrap();
        q.fail(id, t1).unwrap();
        assert_eq!(q.get_state(id), Some(JobState::Dead));
    }

    #[test]
    fn errors_on_invalid_transitions() {
        let mut q = Queue::new(3, Duration::from_millis(10));
        let id = q.enqueue("p".to_string());
        // Can't succeed when not Running.
        assert!(matches!(q.succeed(id), Err(QueueError::NotRunning(_))));
        let unknown = JobId(9999);
        assert!(matches!(
            q.succeed(unknown),
            Err(QueueError::UnknownJob(_))
        ));
        assert!(matches!(
            q.fail(unknown, Instant::now()),
            Err(QueueError::UnknownJob(_))
        ));
    }
}
```
