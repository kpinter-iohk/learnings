expr  := let | if | or
let   := "let" IDENT "=" expr "in" expr
if    := "if" expr "then" expr "else" expr
or    := and ("or" and)*
and   := not ("and" not)*
not   := "not" not | cmp
cmp   := add (("==" | "!=" | "<" | "<=" | ">" | ">=") add)?

mul   := unary (("*" | "/") unary)*
unary := "-" unary | atom
atom  := INT | "true" | "false" | IDENT | "(" expr ")"

fn tokenize(input: &str) -> Result<Vec<(Tok, usize)>, ParseError> {
    let bytes = input.as_bytes();
    let mut tokens = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let c = bytes[i];
        if c == b' ' || c == b'\t' || c == b'\n' || c == b'\r' {
            i += 1;
            continue;
        }
        let start = i;
        if c.is_ascii_digit() {
            let mut j = i;
            while j < bytes.len() && bytes[j].is_ascii_digit() {
                j += 1;
            }
            let s = std::str::from_utf8(&bytes[i..j]).unwrap();
            let n: i64 = s.parse().map_err(|_| ParseError {
                offset: start,
                message: "invalid integer literal".to_string(),
            })?;
            tokens.push((Tok::Int(n), start));
            i = j;
            continue;
        }
        if c.is_ascii_lowercase() || c == b'_' {
            let mut j = i;
            while j < bytes.len()
                && (bytes[j].is_ascii_lowercase() || bytes[j].is_ascii_digit() || bytes[j] == b'_')
            {
                j += 1;
            }
            let s = std::str::from_utf8(&bytes[i..j]).unwrap();
            let tok = match s {
                "let" => Tok::Let,
                "in" => Tok::In,
                "if" => Tok::If,
                "then" => Tok::Then,
                "else" => Tok::Else,
                "or" => Tok::Or,
                "and" => Tok::And,
                "not" => Tok::Not,
                "true" => Tok::True,
                "false" => Tok::False,
                _ => Tok::Ident(s.to_string()),
            };
            tokens.push((tok, start));
            i = j;
            continue;
        }
        match c {
            b'+' => {
                tokens.push((Tok::Plus, start));
                i += 1;
            }
            b'-' => {
                tokens.push((Tok::Minus, start));
                i += 1;
            }
            b'*' => {
                tokens.push((Tok::Star, start));
                i += 1;
            }
            b'/' => {
                tokens.push((Tok::Slash, start));
                i += 1;
            }
            b'(' => {
                tokens.push((Tok::LParen, start));
                i += 1;
            }
            b')' => {
                tokens.push((Tok::RParen, start));
                i += 1;
            }
            b'=' => {
                if i + 1 < bytes.len() && bytes[i + 1] == b'=' {
                    tokens.push((Tok::EqEq, start));
                    i += 2;
                } else {
                    tokens.push((Tok::Eq, start));
                    i += 1;
                }
            }
            b'!' => {
                if i + 1 < bytes.len() && bytes[i + 1] == b'=' {
                    tokens.push((Tok::Ne, start));
                    i += 2;
                } else {
                    return Err(ParseError {
                        offset: start,
                        message: "expected '=' after '!'".to_string(),
                    });
                }
            }
            b'<' => {
                if i + 1 < bytes.len() && bytes[i + 1] == b'=' {
                    tokens.push((Tok::Le, start));
                    i += 2;
                } else {
                    tokens.push((Tok::Lt, start));
                    i += 1;
                }
            }
            b'>' => {
                if i + 1 < bytes.len() && bytes[i + 1] == b'=' {
                    tokens.push((Tok::Ge, start));
                    i += 2;
                } else {
                    tokens.push((Tok::Gt, start));
                    i += 1;
                }
            }
            _ => {
                return Err(ParseError {
                    offset: start,
                    message: format!("unexpected character {:?}", c as char),
                });
            }
        }
    }
    Ok(tokens)
}

struct Parser {
    tokens: Vec<(Tok, usize)>,
    pos: usize,
    eof_offset: usize,
}

impl Parser {
    fn peek(&self) -> Option<&Tok> {
        self.tokens.get(self.pos).map(|(t, _)| t)
    }

    fn peek_offset(&self) -> usize {
        self.tokens
            .get(self.pos)
            .map(|(_, o)| *o)
            .unwrap_or(self.eof_offset)
    }

    fn bump(&mut self) {
        self.pos += 1;
    }

    fn expect<F: Fn(&Tok) -> bool>(&mut self, predicate: F, msg: &str) -> Result<(), ParseError> {
        match self.peek() {
            Some(t) if predicate(t) => {
                self.pos += 1;
                Ok(())
            }
            _ => Err(ParseError {
                offset: self.peek_offset(),
                message: msg.to_string(),
            }),
        }
    }

    fn parse_expr(&mut self) -> Result<Expr, ParseError> {
        match self.peek() {
            Some(Tok::Let) => self.parse_let(),
            Some(Tok::If) => self.parse_if(),
            _ => self.parse_or(),
        }
    }

    fn parse_let(&mut self) -> Result<Expr, ParseError> {
        self.bump();
        let name = match self.tokens.get(self.pos) {
            Some((Tok::Ident(s), _)) => {
                let n = s.clone();
                self.pos += 1;
                n
            }
            _ => {
                return Err(ParseError {
                    offset: self.peek_offset(),
                    message: "expected identifier after 'let'".to_string(),
                });
            }
        };
        self.expect(|t| matches!(t, Tok::Eq), "expected '=' in let")?;
        let val = self.parse_expr()?;
        self.expect(|t| matches!(t, Tok::In), "expected 'in' in let")?;
        let body = self.parse_expr()?;
        Ok(Expr::Let(name, Box::new(val), Box::new(body)))
    }

    fn parse_if(&mut self) -> Result<Expr, ParseError> {
        self.bump();
        let cond = self.parse_expr()?;
        self.expect(|t| matches!(t, Tok::Then), "expected 'then'")?;
        let then_e = self.parse_expr()?;
        self.expect(|t| matches!(t, Tok::Else), "expected 'else'")?;
        let else_e = self.parse_expr()?;
        Ok(Expr::If(Box::new(cond), Box::new(then_e), Box::new(else_e)))
    }

    fn parse_or(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_and()?;
        while matches!(self.peek(), Some(Tok::Or)) {
            self.bump();
            let right = self.parse_and()?;
            left = Expr::BinOp(BinOp::Or, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_and(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_not()?;
        while matches!(self.peek(), Some(Tok::And)) {
            self.bump();
            let right = self.parse_not()?;
            left = Expr::BinOp(BinOp::And, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_not(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek(), Some(Tok::Not)) {
            self.bump();
            let e = self.parse_not()?;
            Ok(Expr::Not(Box::new(e)))
        } else {
            self.parse_cmp()
        }
    }

    fn parse_cmp(&mut self) -> Result<Expr, ParseError> {
        let left = self.parse_add()?;
        let op = match self.peek() {
            Some(Tok::EqEq) => Some(BinOp::Eq),
            Some(Tok::Ne) => Some(BinOp::Ne),
            Some(Tok::Lt) => Some(BinOp::Lt),
            Some(Tok::Le) => Some(BinOp::Le),
            Some(Tok::Gt) => Some(BinOp::Gt),
            Some(Tok::Ge) => Some(BinOp::Ge),
            _ => None,
        };
        if let Some(op) = op {
            self.bump();
            let right = self.parse_add()?;
            Ok(Expr::BinOp(op, Box::new(left), Box::new(right)))
        } else {
            Ok(left)
        }
    }

    fn parse_add(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_mul()?;
        loop {
            let op = match self.peek() {
                Some(Tok::Plus) => BinOp::Add,
                Some(Tok::Minus) => BinOp::Sub,
                _ => break,
            };
            self.bump();
            let right = self.parse_mul()?;
            left = Expr::BinOp(op, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_mul(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_unary()?;
        loop {
            let op = match self.peek() {
                Some(Tok::Star) => BinOp::Mul,
                Some(Tok::Slash) => BinOp::Div,
                _ => break,
            };
            self.bump();
            let right = self.parse_unary()?;
            left = Expr::BinOp(op, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_unary(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek(), Some(Tok::Minus)) {
            self.bump();
            let e = self.parse_unary()?;
            Ok(Expr::Neg(Box::new(e)))
        } else {
            self.parse_atom()
        }
    }

    fn parse_atom(&mut self) -> Result<Expr, ParseError> {
        let offset = self.peek_offset();
        let tok = self.tokens.get(self.pos).cloned();
        match tok {
            Some((Tok::Int(n), _)) => {
                self.pos += 1;
                Ok(Expr::Int(n))
            }
            Some((Tok::True, _)) => {
                self.pos += 1;
                Ok(Expr::Bool(true))
            }
            Some((Tok::False, _)) => {
                self.pos += 1;
                Ok(Expr::Bool(false))
            }
            Some((Tok::Ident(s), _)) => {
                self.pos += 1;
                Ok(Expr::Ident(s))
            }
            Some((Tok::LParen, _)) => {
                self.pos += 1;
                let e = self.parse_expr()?;
                self.expect(|t| matches!(t, Tok::RParen), "expected ')'")?;
                Ok(e)
            }
            _ => Err(ParseError {
                offset,
                message: "expected expression".to_string(),
            }),
        }
    }
}

pub fn parse(input: &str) -> Result<Expr, ParseError> {
    let tokens = tokenize(input)?;
    let eof_offset = input.len();
    let mut parser = Parser {
        tokens,
        pos: 0,
        eof_offset,
    };
    let expr = parser.parse_expr()?;
    if parser.pos < parser.tokens.len() {
        return Err(ParseError {
            offset: parser.peek_offset(),
            message: "unexpected trailing tokens".to_string(),
        });
    }
    Ok(expr)
}

enum Env<'a> {
    Empty,
    Cons(&'a str, Value, &'a Env<'a>),
}

impl<'a> Env<'a> {
    fn lookup(&self, name: &str) -> Option<Value> {
        match self {
            Env::Empty => None,
            Env::Cons(n, v, parent) => {
                if *n == name {
                    Some(v.clone())
                } else {
                    parent.lookup(name)
                }
            }
        }
    }
}

pub fn eval(expr: &Expr) -> Result<Value, EvalError> {
    eval_in(expr, &Env::Empty)
}

fn eval_in(expr: &Expr, env: &Env) -> Result<Value, EvalError> {
    match expr {
        Expr::Int(n) => Ok(Value::Int(*n)),
        Expr::Bool(b) => Ok(Value::Bool(*b)),
        Expr::Ident(name) => env
            .lookup(name)
            .ok_or_else(|| EvalError::UnboundIdent(name.clone())),
        Expr::Neg(e) => match eval_in(e, env)? {
            Value::Int(n) => Ok(Value::Int(n.wrapping_neg())),
            Value::Bool(_) => Err(EvalError::TypeError(
                "unary '-' requires integer".to_string(),
            )),
        },
        Expr::Not(e) => match eval_in(e, env)? {
            Value::Bool(b) => Ok(Value::Bool(!b)),
            Value::Int(_) => Err(EvalError::TypeError("'not' requires boolean".to_string())),
        },
        Expr::If(cond, then_e, else_e) => match eval_in(cond, env)? {
            Value::Bool(true) => eval_in(then_e, env),
            Value::Bool(false) => eval_in(else_e, env),
            Value::Int(_) => Err(EvalError::TypeError(
                "if condition must be boolean".to_string(),
            )),
        },
        Expr::Let(name, val, body) => {
            let v = eval_in(val, env)?;
            let new_env = Env::Cons(name.as_str(), v, env);
            eval_in(body, &new_env)
        }
        Expr::BinOp(op, l, r) => match op {
            BinOp::And => {
                let lv = eval_in(l, env)?;
                let lb = match lv {
                    Value::Bool(b) => b,
                    Value::Int(_) => {
                        return Err(EvalError::TypeError("'and' requires boolean".to_string()));
                    }
                };
                if !lb {
                    return Ok(Value::Bool(false));
                }
                match eval_in(r, env)? {
                    Value::Bool(b) => Ok(Value::Bool(b)),
                    Value::Int(_) => Err(EvalError::TypeError("'and' requires boolean".to_string())),
                }
            }
            BinOp::Or => {
                let lv = eval_in(l, env)?;
                let lb = match lv {
                    Value::Bool(b) => b,
                    Value::Int(_) => {
                        return Err(EvalError::TypeError("'or' requires boolean".to_string()));
                    }
                };
                if lb {
                    return Ok(Value::Bool(true));
                }
                match eval_in(r, env)? {
                    Value::Bool(b) => Ok(Value::Bool(b)),
                    Value::Int(_) => Err(EvalError::TypeError("'or' requires boolean".to_string())),
                }
            }
            _ => {
                let lv = eval_in(l, env)?;
                let rv = eval_in(r, env)?;
                apply_binop(*op, lv, rv)
            }
        },
    }
}

fn apply_binop(op: BinOp, l: Value, r: Value) -> Result<Value, EvalError> {
    use Value::{Bool, Int};
    match (op, l, r) {
        (BinOp::Add, Int(a), Int(b)) => Ok(Int(a.wrapping_add(b))),
        (BinOp::Sub, Int(a), Int(b)) => Ok(Int(a.wrapping_sub(b))),
        (BinOp::Mul, Int(a), Int(b)) => Ok(Int(a.wrapping_mul(b))),
        (BinOp::Div, Int(a), Int(b)) => {
            if b == 0 {
                Err(EvalError::DivisionByZero)
            } else {
                Ok(Int(a.wrapping_div(b)))
            }
        }
        (BinOp::Eq, Int(a), Int(b)) => Ok(Bool(a == b)),
        (BinOp::Eq, Bool(a), Bool(b)) => Ok(Bool(a == b)),
        (BinOp::Ne, Int(a), Int(b)) => Ok(Bool(a != b)),
        (BinOp::Ne, Bool(a), Bool(b)) => Ok(Bool(a != b)),
        (BinOp::Lt, Int(a), Int(b)) => Ok(Bool(a < b)),
        (BinOp::Le, Int(a), Int(b)) => Ok(Bool(a <= b)),
        (BinOp::Gt, Int(a), Int(b)) => Ok(Bool(a > b)),
        (BinOp::Ge, Int(a), Int(b)) => Ok(Bool(a >= b)),
        _ => Err(EvalError::TypeError(
            "operator type mismatch".to_string(),
        )),
    }
}
