The code implements this spec:

---
Implement a tiny expression-language evaluator in Rust.

Grammar:

```
expr  := let | if | or
let   := "let" IDENT "=" expr "in" expr
if    := "if" expr "then" expr "else" expr
or    := and ("or" and)*
and   := not ("and" not)*
not   := "not" not | cmp
cmp   := add (("==" | "!=" | "<" | "<=" | ">" | ">=") add)?
add   := mul (("+" | "-") mul)*
mul   := unary (("*" | "/") unary)*
unary := "-" unary | atom
atom  := INT | "true" | "false" | IDENT | "(" expr ")"
```

Identifiers match `[a-z_][a-z0-9_]*` and are not any of the keywords (`let`, `in`, `if`, `then`, `else`, `or`, `and`, `not`, `true`, `false`). Whitespace separates tokens. No comments.

Values are `i64` or `bool`. There are no implicit conversions: `1 + true` is a runtime error; `if 1 then a else b` is a runtime error.

Public API:

- `pub fn parse(input: &str) -> Result<Expr, ParseError>`
- `pub fn eval(expr: &Expr) -> Result<Value, EvalError>`

The types `Expr`, `Value`, `ParseError`, and `EvalError` must be public.

Additional requirements on those types:

- `Value` must implement `std::fmt::Display` such that integer values print as decimal (e.g., `42`, `-3`) and booleans print as `true` or `false`.
- `ParseError` and `EvalError` must implement `std::fmt::Display` and `std::fmt::Debug`.
- `ParseError` must carry the byte offset within the input where parsing failed.
- `EvalError` must distinguish at least three cases: type errors, division by zero, and unbound identifiers. The exact representation is your choice.

Standard library only. The complete program is one `lib.rs`.

fn token_name(t: &Token) -> String {
    match t {
        Token::Int(n) => format!("integer `{}`", n),
        Token::Ident(s) => format!("identifier `{}`", s),
        Token::Let => "`let`".into(),
        Token::In => "`in`".into(),
        Token::If => "`if`".into(),
        Token::Then => "`then`".into(),

fn tokenize(input: &str) -> Result<Vec<(Token, usize)>, ParseError> {
    let bytes = input.as_bytes();
    let mut pos = 0usize;
    let mut tokens: Vec<(Token, usize)> = Vec::new();

    while pos < bytes.len() {
        let c = bytes[pos];
        if (c as char).is_whitespace() {
            pos += 1;
            continue;
        }
        let start = pos;
        match c {
            b'(' => {
                tokens.push((Token::LParen, start));
                pos += 1;
            }
            b')' => {
                tokens.push((Token::RParen, start));
                pos += 1;
            }
            b'+' => {
                tokens.push((Token::Plus, start));
                pos += 1;
            }
            b'-' => {
                tokens.push((Token::Minus, start));
                pos += 1;
            }
            b'*' => {
                tokens.push((Token::Star, start));
                pos += 1;
            }
            b'/' => {
                tokens.push((Token::Slash, start));
                pos += 1;
            }
            b'=' => {
                if pos + 1 < bytes.len() && bytes[pos + 1] == b'=' {
                    tokens.push((Token::EqEq, start));
                    pos += 2;
                } else {
                    tokens.push((Token::Eq, start));
                    pos += 1;
                }
            }
            b'!' => {
                if pos + 1 < bytes.len() && bytes[pos + 1] == b'=' {
                    tokens.push((Token::Ne, start));
                    pos += 2;
                } else {
                    return Err(ParseError {
                        offset: start,
                        message: "unexpected `!` (did you mean `!=`?)".into(),
                    });
                }
            }
            b'<' => {
                if pos + 1 < bytes.len() && bytes[pos + 1] == b'=' {
                    tokens.push((Token::Le, start));
                    pos += 2;
                } else {
                    tokens.push((Token::Lt, start));
                    pos += 1;
                }
            }
            b'>' => {
                if pos + 1 < bytes.len() && bytes[pos + 1] == b'=' {
                    tokens.push((Token::Ge, start));
                    pos += 2;
                } else {
                    tokens.push((Token::Gt, start));
                    pos += 1;
                }
            }
            b'0'..=b'9' => {
                let mut end = pos;
                while end < bytes.len() && bytes[end].is_ascii_digit() {
                    end += 1;
                }
                let s = std::str::from_utf8(&bytes[start..end]).unwrap();
                let n: i64 = s.parse().map_err(|_| ParseError {
                    offset: start,
                    message: format!("integer literal out of range: {}", s),
                })?;
                tokens.push((Token::Int(n), start));
                pos = end;
            }
            b'a'..=b'z' | b'_' => {
                let mut end = pos;
                while end < bytes.len() {
                    let ch = bytes[end];
                    if ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == b'_' {
                        end += 1;
                    } else {
                        break;
                    }
                }
                let s = std::str::from_utf8(&bytes[start..end]).unwrap();
                let tok = match s {
                    "let" => Token::Let,
                    "in" => Token::In,
                    "if" => Token::If,
                    "then" => Token::Then,
                    "else" => Token::Else,
                    "or" => Token::Or,
                    "and" => Token::And,
                    "not" => Token::Not,
                    "true" => Token::True,
                    "false" => Token::False,
                    other => Token::Ident(other.to_string()),
                };
                tokens.push((tok, start));
                pos = end;
            }
            _ => {
                return Err(ParseError {
                    offset: start,
                    message: format!("unexpected character `{}`", c as char),
                });
            }
        }
    }
    tokens.push((Token::Eof, input.len()));
    Ok(tokens)
}

struct Parser {
    tokens: Vec<(Token, usize)>,
    pos: usize,
}

impl Parser {
    fn new(tokens: Vec<(Token, usize)>) -> Self {
        Parser { tokens, pos: 0 }
    }

    fn peek(&self) -> &Token {
        &self.tokens[self.pos].0
    }

    fn peek_offset(&self) -> usize {
        self.tokens[self.pos].1
    }

    fn eat(&mut self, expected: &Token) -> Result<(), ParseError> {
        if self.peek() == expected {
            self.pos += 1;
            Ok(())
        } else {
            Err(ParseError {
                offset: self.peek_offset(),
                message: format!(
                    "expected {}, found {}",
                    token_name(expected),
                    token_name(self.peek())
                ),
            })
        }
    }

    fn parse_expr(&mut self) -> Result<Expr, ParseError> {
        match self.peek() {
            Token::Let => self.parse_let(),
            Token::If => self.parse_if(),
            _ => self.parse_or(),
        }
    }

    fn parse_let(&mut self) -> Result<Expr, ParseError> {
        self.pos += 1; // consume `let`
        let name = match self.peek().clone() {
            Token::Ident(s) => {
                self.pos += 1;
                s
            }
            other => {
                return Err(ParseError {
                    offset: self.peek_offset(),
                    message: format!("expected identifier after `let`, found {}", token_name(&other)),
                });
            }
        };
        self.eat(&Token::Eq)?;
        let value = self.parse_expr()?;
        self.eat(&Token::In)?;
        let body = self.parse_expr()?;
        Ok(Expr::Let {
            name,
            value: Box::new(value),
            body: Box::new(body),
        })
    }

    fn parse_if(&mut self) -> Result<Expr, ParseError> {
        self.pos += 1; // consume `if`
        let cond = self.parse_expr()?;
        self.eat(&Token::Then)?;
        let then_branch = self.parse_expr()?;
        self.eat(&Token::Else)?;
        let else_branch = self.parse_expr()?;
        Ok(Expr::If {
            cond: Box::new(cond),
            then_branch: Box::new(then_branch),
            else_branch: Box::new(else_branch),
        })
    }

    fn parse_or(&mut self) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_and()?;
        while matches!(self.peek(), Token::Or) {
            self.pos += 1;
            let rhs = self.parse_and()?;
            lhs = Expr::BinOp {
                op: BinOp::Or,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            };
        }
        Ok(lhs)
    }

    fn parse_and(&mut self) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_not()?;
        while matches!(self.peek(), Token::And) {
            self.pos += 1;
            let rhs = self.parse_not()?;
            lhs = Expr::BinOp {
                op: BinOp::And,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            };
        }
        Ok(lhs)
    }

    fn parse_not(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek(), Token::Not) {
            self.pos += 1;
            let inner = self.parse_not()?;
            Ok(Expr::UnaryOp {
                op: UnaryOp::Not,
                expr: Box::new(inner),
            })
        } else {
            self.parse_cmp()
        }
    }

    fn parse_cmp(&mut self) -> Result<Expr, ParseError> {
        let lhs = self.parse_add()?;
        let op = match self.peek() {
            Token::EqEq => Some(BinOp::Eq),
            Token::Ne => Some(BinOp::Ne),
            Token::Lt => Some(BinOp::Lt),
            Token::Le => Some(BinOp::Le),
            Token::Gt => Some(BinOp::Gt),
            Token::Ge => Some(BinOp::Ge),
            _ => None,
        };
        if let Some(op) = op {
            self.pos += 1;
            let rhs = self.parse_add()?;
            Ok(Expr::BinOp {
                op,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            })
        } else {
            Ok(lhs)
        }
    }

    fn parse_add(&mut self) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_mul()?;
        loop {
            let op = match self.peek() {
                Token::Plus => BinOp::Add,
                Token::Minus => BinOp::Sub,
                _ => break,
            };
            self.pos += 1;
            let rhs = self.parse_mul()?;
            lhs = Expr::BinOp {
                op,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            };
        }
        Ok(lhs)
    }

    fn parse_mul(&mut self) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_unary()?;
        loop {
            let op = match self.peek() {
                Token::Star => BinOp::Mul,
                Token::Slash => BinOp::Div,
                _ => break,
            };
            self.pos += 1;
            let rhs = self.parse_unary()?;
            lhs = Expr::BinOp {
                op,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            };
        }
        Ok(lhs)
    }

    fn parse_unary(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek(), Token::Minus) {
            self.pos += 1;
            let inner = self.parse_unary()?;
            Ok(Expr::UnaryOp {
                op: UnaryOp::Neg,
                expr: Box::new(inner),
            })
        } else {
            self.parse_atom()
        }
    }

    fn parse_atom(&mut self) -> Result<Expr, ParseError> {
        let offset = self.peek_offset();
        match self.peek().clone() {
            Token::Int(n) => {
                self.pos += 1;
                Ok(Expr::Int(n))
            }
            Token::True => {
                self.pos += 1;
                Ok(Expr::Bool(true))
            }
            Token::False => {
                self.pos += 1;
                Ok(Expr::Bool(false))
            }
            Token::Ident(s) => {
                self.pos += 1;
                Ok(Expr::Ident(s))
            }
            Token::LParen => {
                self.pos += 1;
                let inner = self.parse_expr()?;
                self.eat(&Token::RParen)?;
                Ok(inner)
            }
            other => Err(ParseError {
                offset,
                message: format!("expected expression, found {}", token_name(&other)),
            }),
        }
    }
}

pub fn parse(input: &str) -> Result<Expr, ParseError> {
    let tokens = tokenize(input)?;
    let mut parser = Parser::new(tokens);
    let expr = parser.parse_expr()?;
    if !matches!(parser.peek(), Token::Eof) {
        return Err(ParseError {
            offset: parser.peek_offset(),
            message: format!("unexpected {} after expression", token_name(parser.peek())),
        });
    }
    Ok(expr)
}

pub fn eval(expr: &Expr) -> Result<Value, EvalError> {
    let mut env: Vec<(String, Value)> = Vec::new();
    eval_in(expr, &mut env)
}

fn lookup(env: &[(String, Value)], name: &str) -> Option<Value> {
    for (n, v) in env.iter().rev() {
        if n == name {
            return Some(*v);
        }
    }
    None
}

fn eval_in(expr: &Expr, env: &mut Vec<(String, Value)>) -> Result<Value, EvalError> {
    match expr {
        Expr::Int(i) => Ok(Value::Int(*i)),
        Expr::Bool(b) => Ok(Value::Bool(*b)),
        Expr::Ident(name) => {
            lookup(env, name).ok_or_else(|| EvalError::UnboundIdent(name.clone()))
        }
        Expr::Let { name, value, body } => {
            let v = eval_in(value, env)?;
            env.push((name.clone(), v));
            let result = eval_in(body, env);
            env.pop();
            result
        }
        Expr::If {
            cond,
            then_branch,
            else_branch,
        } => match eval_in(cond, env)? {
            Value::Bool(true) => eval_in(then_branch, env),
            Value::Bool(false) => eval_in(else_branch, env),
            Value::Int(_) => Err(EvalError::TypeError(
                "condition of `if` must be bool, found int".into(),
            )),
        },
        Expr::UnaryOp { op, expr } => {
            let v = eval_in(expr, env)?;
            match (op, v) {
                (UnaryOp::Neg, Value::Int(i)) => Ok(Value::Int(i.wrapping_neg())),
                (UnaryOp::Neg, Value::Bool(_)) => Err(EvalError::TypeError(
                    "unary `-` requires int, found bool".into(),
                )),
                (UnaryOp::Not, Value::Bool(b)) => Ok(Value::Bool(!b)),
                (UnaryOp::Not, Value::Int(_)) => Err(EvalError::TypeError(
                    "`not` requires bool, found int".into(),
                )),
            }
        }
        Expr::BinOp { op, lhs, rhs } => match op {
            BinOp::Or => {
                let l = eval_in(lhs, env)?;
                let lb = match l {
                    Value::Bool(b) => b,
                    Value::Int(_) => {
                        return Err(EvalError::TypeError(
                            "`or` requires bool on the left, found int".into(),
                        ))
                    }
                };
                if lb {
                    return Ok(Value::Bool(true));
                }
                let r = eval_in(rhs, env)?;
                match r {
                    Value::Bool(b) => Ok(Value::Bool(b)),
                    Value::Int(_) => Err(EvalError::TypeError(
                        "`or` requires bool on the right, found int".into(),
                    )),
                }
            }
            BinOp::And => {
                let l = eval_in(lhs, env)?;
                let lb = match l {
                    Value::Bool(b) => b,
                    Value::Int(_) => {
                        return Err(EvalError::TypeError(
                            "`and` requires bool on the left, found int".into(),
                        ))
                    }
                };
                if !lb {
                    return Ok(Value::Bool(false));
                }
                let r = eval_in(rhs, env)?;
                match r {
                    Value::Bool(b) => Ok(Value::Bool(b)),
                    Value::Int(_) => Err(EvalError::TypeError(
                        "`and` requires bool on the right, found int".into(),
                    )),
                }
            }
            _ => {
                let l = eval_in(lhs, env)?;
                let r = eval_in(rhs, env)?;
                eval_strict_binop(*op, l, r)
            }
        },
    }
}

fn eval_strict_binop(op: BinOp, l: Value, r: Value) -> Result<Value, EvalError> {
    use Value::{Bool, Int};
    match (op, l, r) {
        (BinOp::Add, Int(a), Int(b)) => Ok(Int(a.wrapping_add(b))),
        (BinOp::Sub, Int(a), Int(b)) => Ok(Int(a.wrapping_sub(b))),
        (BinOp::Mul, Int(a), Int(b)) => Ok(Int(a.wrapping_mul(b))),
        (BinOp::Div, Int(_), Int(0)) => Err(EvalError::DivisionByZero),
        (BinOp::Div, Int(a), Int(b)) => Ok(Int(a.wrapping_div(b))),

        (BinOp::Eq, Int(a), Int(b)) => Ok(Bool(a == b)),
        (BinOp::Eq, Bool(a), Bool(b)) => Ok(Bool(a == b)),
        (BinOp::Ne, Int(a), Int(b)) => Ok(Bool(a != b)),
        (BinOp::Ne, Bool(a), Bool(b)) => Ok(Bool(a != b)),

        (BinOp::Lt, Int(a), Int(b)) => Ok(Bool(a < b)),
        (BinOp::Le, Int(a), Int(b)) => Ok(Bool(a <= b)),
        (BinOp::Gt, Int(a), Int(b)) => Ok(Bool(a > b)),
        (BinOp::Ge, Int(a), Int(b)) => Ok(Bool(a >= b)),

        (op, l, r) => Err(EvalError::TypeError(format!(
            "operator {:?} cannot be applied to {} and {}",
            op,
            type_name(&l),
            type_name(&r)
        ))),
    }
}

fn type_name(v: &Value) -> &'static str {
    match v {
        Value::Int(_) => "int",
        Value::Bool(_) => "bool",
    }
}
