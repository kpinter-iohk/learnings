Implement a tiny expression-language evaluator in Rust.

Grammar:

```
expr  := let | if | or
let   := "let" IDENT "=" expr "in" expr
if    := "if" expr "then" expr "else" expr
or    := and ("or" and)*
and   := not ("and" not)*
not   := "not" not | cmp
cmp   := add (("==" | "!=" | "<" | "<=" | ">" | ">=") add)?
add   := mul (("+" | "-") mul)*
mul   := unary (("*" | "/") unary)*
unary := "-" unary | atom
atom  := INT | "true" | "false" | IDENT | "(" expr ")"
```

Identifiers match `[a-z_][a-z0-9_]*` and are not any of the keywords (`let`, `in`, `if`, `then`, `else`, `or`, `and`, `not`, `true`, `false`). Whitespace separates tokens. No comments.

Values are `i64` or `bool`. There are no implicit conversions: `1 + true` is a runtime error; `if 1 then a else b` is a runtime error.

Public API:

- `pub fn parse(input: &str) -> Result<Expr, ParseError>`
- `pub fn eval(expr: &Expr) -> Result<Value, EvalError>`

The types `Expr`, `Value`, `ParseError`, and `EvalError` must be public.

Additional requirements on those types:

- `Value` must implement `std::fmt::Display` such that integer values print as decimal (e.g., `42`, `-3`) and booleans print as `true` or `false`.
- `ParseError` and `EvalError` must implement `std::fmt::Display` and `std::fmt::Debug`.
- `ParseError` must carry the byte offset within the input where parsing failed.
- `EvalError` must distinguish at least three cases: type errors, division by zero, and unbound identifiers. The exact representation is your choice.

/// Error returned by `parse`. Carries the byte offset of the failure.
pub struct ParseError {

#[derive(Debug, Clone, PartialEq)]
enum Tok {
    Int(i64),
    Ident(String),
    Let,
    In,
    If,
    Then,
    Else,
    Or,
    And,
    Not,
    True,
    False,
    Eq,
    Ne,
    Lt,
    Le,
    Gt,
    Ge,
    Plus,
    Minus,
    Star,
    Slash,
    Assign,
    LParen,
    RParen,
    Eof,
}

struct Lexer<'a> {
    input: &'a [u8],
    pos: usize,
}

impl<'a> Lexer<'a> {
    fn new(input: &'a str) -> Self {
        Lexer {
            input: input.as_bytes(),
            pos: 0,
        }
    }

    fn skip_whitespace(&mut self) {
        while self.pos < self.input.len() {
            let c = self.input[self.pos] as char;
            if c.is_whitespace() {
                self.pos += 1;
            } else {
                break;
            }
        }
    }

    fn next_token(&mut self) -> Result<(Tok, usize), ParseError> {
        self.skip_whitespace();
        let start = self.pos;
        if self.pos >= self.input.len() {
            return Ok((Tok::Eof, start));
        }
        let c = self.input[self.pos] as char;

        // Integer literal
        if c.is_ascii_digit() {
            let mut end = self.pos;
            while end < self.input.len() && (self.input[end] as char).is_ascii_digit() {
                end += 1;
            }
            // SAFETY: only ASCII digits in this slice.
            let s = std::str::from_utf8(&self.input[self.pos..end]).unwrap();
            let n = s.parse::<i64>().map_err(|_| ParseError {
                offset: start,
                message: format!("invalid integer literal '{}'", s),
            })?;
            self.pos = end;
            return Ok((Tok::Int(n), start));
        }

        // Identifier / keyword
        if c == '_' || c.is_ascii_lowercase() {
            let mut end = self.pos;
            while end < self.input.len() {
                let ch = self.input[end] as char;
                if ch == '_' || ch.is_ascii_lowercase() || ch.is_ascii_digit() {
                    end += 1;
                } else {
                    break;
                }
            }
            let s = std::str::from_utf8(&self.input[self.pos..end])
                .unwrap()
                .to_string();
            self.pos = end;
            let tok = match s.as_str() {
                "let" => Tok::Let,
                "in" => Tok::In,
                "if" => Tok::If,
                "then" => Tok::Then,
                "else" => Tok::Else,
                "or" => Tok::Or,
                "and" => Tok::And,
                "not" => Tok::Not,
                "true" => Tok::True,
                "false" => Tok::False,
                _ => Tok::Ident(s),
            };
            return Ok((tok, start));
        }

        // Operators / punctuation
        let c2 = if self.pos + 1 < self.input.len() {
            Some(self.input[self.pos + 1] as char)
        } else {
            None
        };
        let (tok, len): (Tok, usize) = match (c, c2) {
            ('=', Some('=')) => (Tok::Eq, 2),
            ('!', Some('=')) => (Tok::Ne, 2),
            ('<', Some('=')) => (Tok::Le, 2),
            ('>', Some('=')) => (Tok::Ge, 2),
            ('<', _) => (Tok::Lt, 1),
            ('>', _) => (Tok::Gt, 1),
            ('=', _) => (Tok::Assign, 1),
            ('+', _) => (Tok::Plus, 1),
            ('-', _) => (Tok::Minus, 1),
            ('*', _) => (Tok::Star, 1),
            ('/', _) => (Tok::Slash, 1),
            ('(', _) => (Tok::LParen, 1),
            (')', _) => (Tok::RParen, 1),
            _ => {
                return Err(ParseError {
                    offset: start,
                    message: format!("unexpected character {:?}", c),
                });
            }
        };
        self.pos += len;
        Ok((tok, start))
    }
}

// Parser
// ============================================================

struct Parser {
    tokens: Vec<(Tok, usize)>,
    pos: usize,
}

impl Parser {
    fn new(input: &str) -> Result<Self, ParseError> {
        let mut lexer = Lexer::new(input);
        let mut tokens = Vec::new();
        loop {
            let (tok, off) = lexer.next_token()?;
            let is_eof = matches!(tok, Tok::Eof);
            tokens.push((tok, off));
            if is_eof {
                break;
            }
        }
        Ok(Parser { tokens, pos: 0 })
    }

    fn peek(&self) -> &Tok {
        &self.tokens[self.pos].0
    }

    fn peek_offset(&self) -> usize {
        self.tokens[self.pos].1
    }

    fn bump(&mut self) {
        if !matches!(self.peek(), Tok::Eof) {
            self.pos += 1;
        }
    }

    // expr := let | if | or
    fn parse_expr(&mut self) -> Result<Expr, ParseError> {
        match self.peek() {
            Tok::Let => self.parse_let(),
            Tok::If => self.parse_if(),
            _ => self.parse_or(),
        }
    }

    // let := "let" IDENT "=" expr "in" expr
    fn parse_let(&mut self) -> Result<Expr, ParseError> {
        // consume "let"
        self.bump();

        let name = match self.peek().clone() {
            Tok::Ident(n) => {
                self.bump();
                n
            }
            _ => {
                return Err(ParseError {
                    offset: self.peek_offset(),
                    message: "expected identifier after 'let'".to_string(),
                });
            }
        };

        match self.peek() {
            Tok::Assign => self.bump(),
            _ => {
                return Err(ParseError {
                    offset: self.peek_offset(),
                    message: "expected '=' after identifier in 'let'".to_string(),
                });
            }
        }

        let value = Box::new(self.parse_expr()?);

        match self.peek() {
            Tok::In => self.bump(),
            _ => {
                return Err(ParseError {
                    offset: self.peek_offset(),
                    message: "expected 'in' in 'let' expression".to_string(),
                });
            }
        }

        let body = Box::new(self.parse_expr()?);
        Ok(Expr::Let { name, value, body })
    }

    // if := "if" expr "then" expr "else" expr
    fn parse_if(&mut self) -> Result<Expr, ParseError> {
        // consume "if"
        self.bump();
        let cond = Box::new(self.parse_expr()?);

        match self.peek() {
            Tok::Then => self.bump(),
            _ => {
                return Err(ParseError {
                    offset: self.peek_offset(),
                    message: "expected 'then'".to_string(),
                });
            }
        }

        let then_branch = Box::new(self.parse_expr()?);

        match self.peek() {
            Tok::Else => self.bump(),
            _ => {
                return Err(ParseError {
                    offset: self.peek_offset(),
                    message: "expected 'else'".to_string(),
                });
            }
        }

        let else_branch = Box::new(self.parse_expr()?);
        Ok(Expr::If {
            cond,
            then_branch,
            else_branch,
        })
    }

    // or := and ("or" and)*
    fn parse_or(&mut self) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_and()?;
        while matches!(self.peek(), Tok::Or) {
            self.bump();
            let rhs = self.parse_and()?;
            lhs = Expr::Binary {
                op: BinOp::Or,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            };
        }
        Ok(lhs)
    }

    // and := not ("and" not)*
    fn parse_and(&mut self) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_not()?;
        while matches!(self.peek(), Tok::And) {
            self.bump();
            let rhs = self.parse_not()?;
            lhs = Expr::Binary {
                op: BinOp::And,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            };
        }
        Ok(lhs)
    }

    // not := "not" not | cmp
    fn parse_not(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek(), Tok::Not) {
            self.bump();
            let inner = self.parse_not()?;
            Ok(Expr::Unary {
                op: UnaryOp::Not,
                expr: Box::new(inner),
            })
        } else {
            self.parse_cmp()
        }
    }

    // cmp := add (("==" | "!=" | "<" | "<=" | ">" | ">=") add)?
    fn parse_cmp(&mut self) -> Result<Expr, ParseError> {
        let lhs = self.parse_add()?;
        let op = match self.peek() {
            Tok::Eq => Some(BinOp::Eq),
            Tok::Ne => Some(BinOp::Ne),
            Tok::Lt => Some(BinOp::Lt),
            Tok::Le => Some(BinOp::Le),
            Tok::Gt => Some(BinOp::Gt),
            Tok::Ge => Some(BinOp::Ge),
            _ => None,
        };
        if let Some(op) = op {
            self.bump();
            let rhs = self.parse_add()?;
            Ok(Expr::Binary {
                op,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            })
        } else {
            Ok(lhs)
        }
    }

    // add := mul (("+" | "-") mul)*
    fn parse_add(&mut self) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_mul()?;
        loop {
            let op = match self.peek() {
                Tok::Plus => BinOp::Add,
                Tok::Minus => BinOp::Sub,
                _ => break,
            };
            self.bump();
            let rhs = self.parse_mul()?;
            lhs = Expr::Binary {
                op,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            };
        }
        Ok(lhs)
    }

    // mul := unary (("*" | "/") unary)*
    fn parse_mul(&mut self) -> Result<Expr, ParseError> {
        let mut lhs = self.parse_unary()?;
        loop {
            let op = match self.peek() {
                Tok::Star => BinOp::Mul,
                Tok::Slash => BinOp::Div,
                _ => break,
            };
            self.bump();
            let rhs = self.parse_unary()?;
            lhs = Expr::Binary {
                op,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
            };
        }
        Ok(lhs)
    }

    // unary := "-" unary | atom
    fn parse_unary(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek(), Tok::Minus) {
            self.bump();
            let inner = self.parse_unary()?;
            Ok(Expr::Unary {
                op: UnaryOp::Neg,
                expr: Box::new(inner),
            })
        } else {
            self.parse_atom()
        }
    }

    // atom := INT | "true" | "false" | IDENT | "(" expr ")"
    fn parse_atom(&mut self) -> Result<Expr, ParseError> {
        let offset = self.peek_offset();
        match self.peek().clone() {
            Tok::Int(n) => {
                self.bump();
                Ok(Expr::Int(n))
            }
            Tok::True => {
                self.bump();
                Ok(Expr::Bool(true))
            }
            Tok::False => {
                self.bump();
                Ok(Expr::Bool(false))
            }
            Tok::Ident(name) => {
                self.bump();
                Ok(Expr::Ident(name))
            }
            Tok::LParen => {
                self.bump();
                let e = self.parse_expr()?;
                match self.peek() {
                    Tok::RParen => {
                        self.bump();
                        Ok(e)
                    }
                    _ => Err(ParseError {
                        offset: self.peek_offset(),
                        message: "expected ')'".to_string(),
                    }),
                }
            }
            _ => Err(ParseError {
                offset,
                message: "expected expression".to_string(),
            }),
        }
    }
}

// ============================================================
// Public entry points
// ============================================================

pub fn parse(input: &str) -> Result<Expr, ParseError> {
    let mut parser = Parser::new(input)?;
    let expr = parser.parse_expr()?;
    if !matches!(parser.peek(), Tok::Eof) {
        return Err(ParseError {
            offset: parser.peek_offset(),
            message: "unexpected trailing input".to_string(),
        });
    }
    Ok(expr)
}

pub fn eval(expr: &Expr) -> Result<Value, EvalError> {
    let mut env: Vec<(String, Value)> = Vec::new();
    eval_in(expr, &mut env)
}

// ============================================================
// Evaluator
// ============================================================

fn lookup(env: &[(String, Value)], name: &str) -> Option<Value> {
    for (n, v) in env.iter().rev() {
        if n == name {
            return Some(v.clone());
        }
    }
    None
}

fn eval_in(expr: &Expr, env: &mut Vec<(String, Value)>) -> Result<Value, EvalError> {
    match expr {
        Expr::Int(n) => Ok(Value::Int(*n)),
        Expr::Bool(b) => Ok(Value::Bool(*b)),
        Expr::Ident(name) => {
            lookup(env, name).ok_or_else(|| EvalError::UnboundIdentifier(name.clone()))
        }
        Expr::Let { name, value, body } => {
            let v = eval_in(value, env)?;
            env.push((name.clone(), v));
            let result = eval_in(body, env);
            env.pop();
            result
        }
        Expr::If {
            cond,
            then_branch,
            else_branch,
        } => match eval_in(cond, env)? {
            Value::Bool(true) => eval_in(then_branch, env),
            Value::Bool(false) => eval_in(else_branch, env),
            Value::Int(_) => Err(EvalError::TypeError(
                "condition of 'if' must be a bool".to_string(),
            )),
        },
        Expr::Unary { op, expr } => {
            let v = eval_in(expr, env)?;
            match op {
                UnaryOp::Neg => match v {
                    Value::Int(n) => Ok(Value::Int(n.wrapping_neg())),
                    Value::Bool(_) => Err(EvalError::TypeError(
                        "unary '-' requires an int operand".to_string(),
                    )),
                },
                UnaryOp::Not => match v {
                    Value::Bool(b) => Ok(Value::Bool(!b)),
                    Value::Int(_) => Err(EvalError::TypeError(
                        "'not' requires a bool operand".to_string(),
                    )),
                },
            }
        }
        Expr::Binary { op, lhs, rhs } => match op {
            // Short-circuit evaluation for boolean connectives.
            BinOp::And => {
                let l = eval_in(lhs, env)?;
                let lb = match l {
                    Value::Bool(b) => b,
                    Value::Int(_) => {
                        return Err(EvalError::TypeError(
                            "'and' requires bool operands".to_string(),
                        ));
                    }
                };
                if !lb {
                    return Ok(Value::Bool(false));
                }
                match eval_in(rhs, env)? {
                    Value::Bool(b) => Ok(Value::Bool(b)),
                    Value::Int(_) => Err(EvalError::TypeError(
                        "'and' requires bool operands".to_string(),
                    )),
                }
            }
            BinOp::Or => {
                let l = eval_in(lhs, env)?;
                let lb = match l {
                    Value::Bool(b) => b,
                    Value::Int(_) => {
                        return Err(EvalError::TypeError(
                            "'or' requires bool operands".to_string(),
                        ));
                    }
                };
                if lb {
                    return Ok(Value::Bool(true));
                }
                match eval_in(rhs, env)? {
                    Value::Bool(b) => Ok(Value::Bool(b)),
                    Value::Int(_) => Err(EvalError::TypeError(
                        "'or' requires bool operands".to_string(),
                    )),
                }
            }
            // Strict binary operators.
            _ => {
                let l = eval_in(lhs, env)?;
                let r = eval_in(rhs, env)?;
                eval_strict_binop(op, l, r)
            }
        },
    }
}

fn eval_strict_binop(op: &BinOp, l: Value, r: Value) -> Result<Value, EvalError> {
    match op {
        BinOp::Add | BinOp::Sub | BinOp::Mul | BinOp::Div => match (l, r) {
            (Value::Int(a), Value::Int(b)) => match op {
                BinOp::Add => Ok(Value::Int(a.wrapping_add(b))),
                BinOp::Sub => Ok(Value::Int(a.wrapping_sub(b))),
                BinOp::Mul => Ok(Value::Int(a.wrapping_mul(b))),
                BinOp::Div => {
                    if b == 0 {
                        Err(EvalError::DivisionByZero)
                    } else {
                        Ok(Value::Int(a.wrapping_div(b)))
                    }
                }
                _ => unreachable!(),
            },
            _ => Err(EvalError::TypeError(
                "arithmetic operators require int operands".to_string(),
            )),
        },
        BinOp::Eq | BinOp::Ne => match (l, r) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Bool(if matches!(op, BinOp::Eq) {
                a == b
            } else {
                a != b
            })),
            (Value::Bool(a), Value::Bool(b)) => Ok(Value::Bool(if matches!(op, BinOp::Eq) {
                a == b
            } else {
                a != b
            })),
            _ => Err(EvalError::TypeError(
                "equality requires operands of the same type".to_string(),
            )),
        },
        BinOp::Lt | BinOp::Le | BinOp::Gt | BinOp::Ge => match (l, r) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Bool(match op {
                BinOp::Lt => a < b,
                BinOp::Le => a <= b,
                BinOp::Gt => a > b,
                BinOp::Ge => a >= b,
                _ => unreachable!(),
            })),
            _ => Err(EvalError::TypeError(
                "ordering comparisons require int operands".to_string(),
            )),
        },
        BinOp::And | BinOp::Or => unreachable!("handled with short-circuiting"),
    }
}
