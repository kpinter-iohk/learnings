```rust
//! Tiny expression-language evaluator.
//!
//! Single-file `lib.rs`. Standard library only.

use std::fmt;

// ---------------------------------------------------------------------------
// Public AST and value types
// ---------------------------------------------------------------------------

#[derive(Debug, Clone)]
pub enum Expr {
    Int(i64),
    Bool(bool),
    Ident(String),
    Let(String, Box<Expr>, Box<Expr>),
    If(Box<Expr>, Box<Expr>, Box<Expr>),
    BinOp(BinOp, Box<Expr>, Box<Expr>),
    UnaryOp(UnaryOp, Box<Expr>),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum BinOp {
    Or,
    And,
    Eq,
    NotEq,
    Lt,
    LtEq,
    Gt,
    GtEq,
    Add,
    Sub,
    Mul,
    Div,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum UnaryOp {
    Not,
    Neg,
}

#[derive(Clone, Copy, PartialEq, Eq)]
pub enum Value {
    Int(i64),
    Bool(bool),
}

impl fmt::Display for Value {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Value::Int(n) => write!(f, "{}", n),
            Value::Bool(b) => write!(f, "{}", b),
        }
    }
}

impl fmt::Debug for Value {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Value::Int(n) => write!(f, "Int({})", n),
            Value::Bool(b) => write!(f, "Bool({})", b),
        }
    }
}

// ---------------------------------------------------------------------------
// Errors
// ---------------------------------------------------------------------------

#[derive(Clone)]
pub struct ParseError {
    pub offset: usize,
    pub message: String,
}

impl fmt::Display for ParseError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "parse error at offset {}: {}", self.offset, self.message)
    }
}

impl fmt::Debug for ParseError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(
            f,
            "ParseError {{ offset: {}, message: {:?} }}",
            self.offset, self.message
        )
    }
}

#[derive(Clone)]
pub enum EvalError {
    TypeError(String),
    DivisionByZero,
    UnboundIdentifier(String),
}

impl fmt::Display for EvalError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            EvalError::TypeError(msg) => write!(f, "type error: {}", msg),
            EvalError::DivisionByZero => write!(f, "division by zero"),
            EvalError::UnboundIdentifier(name) => write!(f, "unbound identifier: {}", name),
        }
    }
}

impl fmt::Debug for EvalError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            EvalError::TypeError(msg) => write!(f, "TypeError({:?})", msg),
            EvalError::DivisionByZero => write!(f, "DivisionByZero"),
            EvalError::UnboundIdentifier(name) => write!(f, "UnboundIdentifier({:?})", name),
        }
    }
}

// ---------------------------------------------------------------------------
// Lexer
// ---------------------------------------------------------------------------

#[derive(Debug, Clone)]
enum Tok {
    Int(i64),
    Ident(String),
    // keywords
    Let,
    In,
    If,
    Then,
    Else,
    Or,
    And,
    Not,
    True,
    False,
    // operators / punctuation
    EqEq,
    BangEq,
    Lt,
    LtEq,
    Gt,
    GtEq,
    Plus,
    Minus,
    Star,
    Slash,
    Assign,
    LParen,
    RParen,
}

#[derive(Debug, Clone)]
struct Token {
    tok: Tok,
    offset: usize,
}

fn is_ident_start(b: u8) -> bool {
    b == b'_' || (b'a'..=b'z').contains(&b)
}

fn is_ident_cont(b: u8) -> bool {
    is_ident_start(b) || b.is_ascii_digit()
}

fn tokenize(input: &str) -> Result<Vec<Token>, ParseError> {
    let bytes = input.as_bytes();
    let mut tokens = Vec::new();
    let mut i = 0usize;

    while i < bytes.len() {
        let c = bytes[i];

        // whitespace
        if c == b' ' || c == b'\t' || c == b'\n' || c == b'\r' {
            i += 1;
            continue;
        }

        let start = i;

        // integer literal
        if c.is_ascii_digit() {
            let mut end = i;
            while end < bytes.len() && bytes[end].is_ascii_digit() {
                end += 1;
            }
            // reject identifier characters immediately after digits (e.g., `1a`)
            if end < bytes.len() && is_ident_start(bytes[end]) {
                return Err(ParseError {
                    offset: end,
                    message: format!(
                        "unexpected character {:?} after integer literal",
                        bytes[end] as char
                    ),
                });
            }
            let s = &input[i..end];
            let n: i64 = s.parse().map_err(|_| ParseError {
                offset: start,
                message: format!("invalid integer literal: {}", s),
            })?;
            tokens.push(Token {
                tok: Tok::Int(n),
                offset: start,
            });
            i = end;
            continue;
        }

        // identifier or keyword
        if is_ident_start(c) {
            let mut end = i;
            while end < bytes.len() && is_ident_cont(bytes[end]) {
                end += 1;
            }
            let s = &input[i..end];
            let tok = match s {
                "let" => Tok::Let,
                "in" => Tok::In,
                "if" => Tok::If,
                "then" => Tok::Then,
                "else" => Tok::Else,
                "or" => Tok::Or,
                "and" => Tok::And,
                "not" => Tok::Not,
                "true" => Tok::True,
                "false" => Tok::False,
                _ => Tok::Ident(s.to_string()),
            };
            tokens.push(Token { tok, offset: start });
            i = end;
            continue;
        }

        // two-character operators
        if i + 1 < bytes.len() {
            let pair = (bytes[i], bytes[i + 1]);
            let two = match pair {
                (b'=', b'=') => Some(Tok::EqEq),
                (b'!', b'=') => Some(Tok::BangEq),
                (b'<', b'=') => Some(Tok::LtEq),
                (b'>', b'=') => Some(Tok::GtEq),
                _ => None,
            };
            if let Some(t) = two {
                tokens.push(Token {
                    tok: t,
                    offset: start,
                });
                i += 2;
                continue;
            }
        }

        // single-character operators
        let single = match c {
            b'<' => Tok::Lt,
            b'>' => Tok::Gt,
            b'+' => Tok::Plus,
            b'-' => Tok::Minus,
            b'*' => Tok::Star,
            b'/' => Tok::Slash,
            b'=' => Tok::Assign,
            b'(' => Tok::LParen,
            b')' => Tok::RParen,
            _ => {
                return Err(ParseError {
                    offset: start,
                    message: format!("unexpected character {:?}", c as char),
                });
            }
        };
        tokens.push(Token {
            tok: single,
            offset: start,
        });
        i += 1;
    }

    Ok(tokens)
}

// ---------------------------------------------------------------------------
// Parser (recursive descent)
// ---------------------------------------------------------------------------

struct Parser {
    tokens: Vec<Token>,
    pos: usize,
    end_offset: usize,
}

impl Parser {
    fn peek(&self) -> Option<&Tok> {
        self.tokens.get(self.pos).map(|t| &t.tok)
    }

    fn current_offset(&self) -> usize {
        self.tokens
            .get(self.pos)
            .map(|t| t.offset)
            .unwrap_or(self.end_offset)
    }

    fn bump(&mut self) -> Option<Token> {
        if self.pos < self.tokens.len() {
            let t = self.tokens[self.pos].clone();
            self.pos += 1;
            Some(t)
        } else {
            None
        }
    }

    fn eat(&mut self, expected: &Tok) -> bool {
        if let Some(t) = self.peek() {
            if std::mem::discriminant(t) == std::mem::discriminant(expected) {
                self.pos += 1;
                return true;
            }
        }
        false
    }

    fn expect(&mut self, expected: &Tok, what: &str) -> Result<(), ParseError> {
        let off = self.current_offset();
        if self.eat(expected) {
            Ok(())
        } else {
            Err(ParseError {
                offset: off,
                message: format!("expected {}", what),
            })
        }
    }

    fn parse_expr(&mut self) -> Result<Expr, ParseError> {
        match self.peek() {
            Some(Tok::Let) => self.parse_let(),
            Some(Tok::If) => self.parse_if(),
            _ => self.parse_or(),
        }
    }

    fn parse_let(&mut self) -> Result<Expr, ParseError> {
        // 'let' has been peeked
        self.bump();

        let off = self.current_offset();
        let name = match self.bump() {
            Some(Token {
                tok: Tok::Ident(n), ..
            }) => n,
            _ => {
                return Err(ParseError {
                    offset: off,
                    message: "expected identifier after 'let'".to_string(),
                });
            }
        };

        self.expect(&Tok::Assign, "'=' in let-binding")?;
        let value = self.parse_expr()?;
        self.expect(&Tok::In, "'in' in let-binding")?;
        let body = self.parse_expr()?;

        Ok(Expr::Let(name, Box::new(value), Box::new(body)))
    }

    fn parse_if(&mut self) -> Result<Expr, ParseError> {
        self.bump(); // 'if'
        let cond = self.parse_expr()?;
        self.expect(&Tok::Then, "'then'")?;
        let then_e = self.parse_expr()?;
        self.expect(&Tok::Else, "'else'")?;
        let else_e = self.parse_expr()?;
        Ok(Expr::If(
            Box::new(cond),
            Box::new(then_e),
            Box::new(else_e),
        ))
    }

    fn parse_or(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_and()?;
        while matches!(self.peek(), Some(Tok::Or)) {
            self.bump();
            let right = self.parse_and()?;
            left = Expr::BinOp(BinOp::Or, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_and(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_not()?;
        while matches!(self.peek(), Some(Tok::And)) {
            self.bump();
            let right = self.parse_not()?;
            left = Expr::BinOp(BinOp::And, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_not(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek(), Some(Tok::Not)) {
            self.bump();
            let inner = self.parse_not()?;
            Ok(Expr::UnaryOp(UnaryOp::Not, Box::new(inner)))
        } else {
            self.parse_cmp()
        }
    }

    fn parse_cmp(&mut self) -> Result<Expr, ParseError> {
        let left = self.parse_add()?;
        let op = match self.peek() {
            Some(Tok::EqEq) => Some(BinOp::Eq),
            Some(Tok::BangEq) => Some(BinOp::NotEq),
            Some(Tok::Lt) => Some(BinOp::Lt),
            Some(Tok::LtEq) => Some(BinOp::LtEq),
            Some(Tok::Gt) => Some(BinOp::Gt),
            Some(Tok::GtEq) => Some(BinOp::GtEq),
            _ => None,
        };
        if let Some(op) = op {
            self.bump();
            let right = self.parse_add()?;
            Ok(Expr::BinOp(op, Box::new(left), Box::new(right)))
        } else {
            Ok(left)
        }
    }

    fn parse_add(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_mul()?;
        loop {
            let op = match self.peek() {
                Some(Tok::Plus) => BinOp::Add,
                Some(Tok::Minus) => BinOp::Sub,
                _ => break,
            };
            self.bump();
            let right = self.parse_mul()?;
            left = Expr::BinOp(op, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_mul(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_unary()?;
        loop {
            let op = match self.peek() {
                Some(Tok::Star) => BinOp::Mul,
                Some(Tok::Slash) => BinOp::Div,
                _ => break,
            };
            self.bump();
            let right = self.parse_unary()?;
            left = Expr::BinOp(op, Box::new(left), Box::new(right));
        }
        Ok(left)
    }

    fn parse_unary(&mut self) -> Result<Expr, ParseError> {
        if matches!(self.peek(), Some(Tok::Minus)) {
            self.bump();
            let inner = self.parse_unary()?;
            Ok(Expr::UnaryOp(UnaryOp::Neg, Box::new(inner)))
        } else {
            self.parse_atom()
        }
    }

    fn parse_atom(&mut self) -> Result<Expr, ParseError> {
        let off = self.current_offset();
        match self.bump() {
            Some(Token {
                tok: Tok::Int(n), ..
            }) => Ok(Expr::Int(n)),
            Some(Token {
                tok: Tok::True, ..
            }) => Ok(Expr::Bool(true)),
            Some(Token {
                tok: Tok::False, ..
            }) => Ok(Expr::Bool(false)),
            Some(Token {
                tok: Tok::Ident(name),
                ..
            }) => Ok(Expr::Ident(name)),
            Some(Token {
                tok: Tok::LParen, ..
            }) => {
                let e = self.parse_expr()?;
                let close_off = self.current_offset();
                if !self.eat(&Tok::RParen) {
                    return Err(ParseError {
                        offset: close_off,
                        message: "expected ')'".to_string(),
                    });
                }
                Ok(e)
            }
            None => Err(ParseError {
                offset: off,
                message: "unexpected end of input".to_string(),
            }),
            Some(t) => Err(ParseError {
                offset: t.offset,
                message: format!("unexpected token in expression"),
            }),
        }
    }
}

pub fn parse(input: &str) -> Result<Expr, ParseError> {
    let tokens = tokenize(input)?;
    let mut parser = Parser {
        tokens,
        pos: 0,
        end_offset: input.len(),
    };
    let expr = parser.parse_expr()?;
    if parser.pos < parser.tokens.len() {
        return Err(ParseError {
            offset: parser.tokens[parser.pos].offset,
            message: "unexpected trailing input".to_string(),
        });
    }
    Ok(expr)
}

// ---------------------------------------------------------------------------
// Evaluator
// ---------------------------------------------------------------------------

pub fn eval(expr: &Expr) -> Result<Value, EvalError> {
    let env: Vec<(&str, Value)> = Vec::new();
    eval_in(expr, &env)
}

fn lookup<'a>(env: &'a [(&'a str, Value)], name: &str) -> Option<Value> {
    for (k, v) in env.iter().rev() {
        if *k == name {
            return Some(*v);
        }
    }
    None
}

fn eval_in<'a>(expr: &'a Expr, env: &[(&'a str, Value)]) -> Result<Value, EvalError> {
    match expr {
        Expr::Int(n) => Ok(Value::Int(*n)),
        Expr::Bool(b) => Ok(Value::Bool(*b)),
        Expr::Ident(name) => {
            lookup(env, name).ok_or_else(|| EvalError::UnboundIdentifier(name.clone()))
        }
        Expr::Let(name, value, body) => {
            let v = eval_in(value, env)?;
            let mut new_env: Vec<(&str, Value)> = env.to_vec();
            new_env.push((name.as_str(), v));
            eval_in(body, &new_env)
        }
        Expr::If(cond, then_e, else_e) => match eval_in(cond, env)? {
            Value::Bool(true) => eval_in(then_e, env),
            Value::Bool(false) => eval_in(else_e, env),
            Value::Int(_) => Err(EvalError::TypeError(
                "condition of 'if' must be a bool".to_string(),
            )),
        },
        Expr::UnaryOp(op, inner) => {
            let v = eval_in(inner, env)?;
            match (op, v) {
                (UnaryOp::Neg, Value::Int(n)) => Ok(Value::Int(n.wrapping_neg())),
                (UnaryOp::Not, Value::Bool(b)) => Ok(Value::Bool(!b)),
                (UnaryOp::Neg, _) => Err(EvalError::TypeError(
                    "unary '-' requires an integer operand".to_string(),
                )),
                (UnaryOp::Not, _) => Err(EvalError::TypeError(
                    "'not' requires a bool operand".to_string(),
                )),
            }
        }
        Expr::BinOp(op, l, r) => eval_binop(*op, l, r, env),
    }
}

fn eval_binop<'a>(
    op: BinOp,
    l: &'a Expr,
    r: &'a Expr,
    env: &[(&'a str, Value)],
) -> Result<Value, EvalError> {
    // Short-circuit logical operators.
    match op {
        BinOp::And => {
            let lv = eval_in(l, env)?;
            return match lv {
                Value::Bool(false) => Ok(Value::Bool(false)),
                Value::Bool(true) => match eval_in(r, env)? {
                    Value::Bool(b) => Ok(Value::Bool(b)),
                    _ => Err(EvalError::TypeError(
                        "'and' requires bool operands".to_string(),
                    )),
                },
                _ => Err(EvalError::TypeError(
                    "'and' requires bool operands".to_string(),
                )),
            };
        }
        BinOp::Or => {
            let lv = eval_in(l, env)?;
            return match lv {
                Value::Bool(true) => Ok(Value::Bool(true)),
                Value::Bool(false) => match eval_in(r, env)? {
                    Value::Bool(b) => Ok(Value::Bool(b)),
                    _ => Err(EvalError::TypeError(
                        "'or' requires bool operands".to_string(),
                    )),
                },
                _ => Err(EvalError::TypeError(
                    "'or' requires bool operands".to_string(),
                )),
            };
        }
        _ => {}
    }

    let lv = eval_in(l, env)?;
    let rv = eval_in(r, env)?;

    match op {
        BinOp::Add => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Int(a.wrapping_add(b))),
            _ => Err(EvalError::TypeError(
                "'+' requires integer operands".to_string(),
            )),
        },
        BinOp::Sub => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Int(a.wrapping_sub(b))),
            _ => Err(EvalError::TypeError(
                "'-' requires integer operands".to_string(),
            )),
        },
        BinOp::Mul => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Int(a.wrapping_mul(b))),
            _ => Err(EvalError::TypeError(
                "'*' requires integer operands".to_string(),
            )),
        },
        BinOp::Div => match (lv, rv) {
            (Value::Int(_), Value::Int(0)) => Err(EvalError::DivisionByZero),
            (Value::Int(a), Value::Int(b)) => Ok(Value::Int(a.wrapping_div(b))),
            _ => Err(EvalError::TypeError(
                "'/' requires integer operands".to_string(),
            )),
        },
        BinOp::Eq => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Bool(a == b)),
            (Value::Bool(a), Value::Bool(b)) => Ok(Value::Bool(a == b)),
            _ => Err(EvalError::TypeError(
                "'==' requires operands of the same type".to_string(),
            )),
        },
        BinOp::NotEq => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Bool(a != b)),
            (Value::Bool(a), Value::Bool(b)) => Ok(Value::Bool(a != b)),
            _ => Err(EvalError::TypeError(
                "'!=' requires operands of the same type".to_string(),
            )),
        },
        BinOp::Lt => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Bool(a < b)),
            _ => Err(EvalError::TypeError(
                "'<' requires integer operands".to_string(),
            )),
        },
        BinOp::LtEq => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Bool(a <= b)),
            _ => Err(EvalError::TypeError(
                "'<=' requires integer operands".to_string(),
            )),
        },
        BinOp::Gt => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Bool(a > b)),
            _ => Err(EvalError::TypeError(
                "'>' requires integer operands".to_string(),
            )),
        },
        BinOp::GtEq => match (lv, rv) {
            (Value::Int(a), Value::Int(b)) => Ok(Value::Bool(a >= b)),
            _ => Err(EvalError::TypeError(
                "'>=' requires integer operands".to_string(),
            )),
        },
        BinOp::And | BinOp::Or => unreachable!("handled above"),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn run(src: &str) -> Result<Value, String> {
        let e = parse(src).map_err(|e| e.to_string())?;
        eval(&e).map_err(|e| e.to_string())
    }

    #[test]
    fn integers_and_arithmetic() {
        assert_eq!(run("1 + 2 * 3").unwrap(), Value::Int(7));
        assert_eq!(run("(1 + 2) * 3").unwrap(), Value::Int(9));
        assert_eq!(run("10 / 3").unwrap(), Value::Int(3));
        assert_eq!(run("- - 5").unwrap(), Value::Int(5));
    }

    #[test]
    fn booleans_and_logic() {
        assert_eq!(run("true and false").unwrap(), Value::Bool(false));
        assert_eq!(run("true or false").unwrap(), Value::Bool(true));
        assert_eq!(run("not true").unwrap(), Value::Bool(false));
        assert_eq!(run("not not false").unwrap(), Value::Bool(false));
    }

    #[test]
    fn comparisons() {
        assert_eq!(run("1 == 1").unwrap(), Value::Bool(true));
        assert_eq!(run("1 != 2").unwrap(), Value::Bool(true));
        assert_eq!(run("1 < 2").unwrap(), Value::Bool(true));
        assert_eq!(run("2 <= 2").unwrap(), Value::Bool(true));
        assert_eq!(run("3 > 2").unwrap(), Value::Bool(true));
        assert_eq!(run("3 >= 3").unwrap(), Value::Bool(true));
        assert_eq!(run("true == true").unwrap(), Value::Bool(true));
    }

    #[test]
    fn let_and_if() {
        assert_eq!(
            run("let x = 1 + 2 in x * x").unwrap(),
            Value::Int(9)
        );
        assert_eq!(
            run("let x = 1 in let x = 2 in x").unwrap(),
            Value::Int(2)
        );
        assert_eq!(
            run("if 1 < 2 then 10 else 20").unwrap(),
            Value::Int(10)
        );
        assert_eq!(
            run("if false then 10 else 20").unwrap(),
            Value::Int(20)
        );
    }

    #[test]
    fn type_errors() {
        assert!(matches!(
            eval(&parse("1 + true").unwrap()),
            Err(EvalError::TypeError(_))
        ));
        assert!(matches!(
            eval(&parse("if 1 then 2 else 3").unwrap()),
            Err(EvalError::TypeError(_))
        ));
    }

    #[test]
    fn division_by_zero() {
        assert!(matches!(
            eval(&parse("10 / 0").unwrap()),
            Err(EvalError::DivisionByZero)
        ));
    }

    #[test]
    fn unbound_identifier() {
        assert!(matches!(
            eval(&parse("x + 1").unwrap()),
            Err(EvalError::UnboundIdentifier(_))
        ));
    }

    #[test]
    fn parse_error_carries_offset() {
        let err = parse("1 + ").unwrap_err();
        assert_eq!(err.offset, 4);
        let err = parse("(1 + 2").unwrap_err();
        // expect ')' at end of input
        assert_eq!(err.offset, 6);
    }

    #[test]
    fn display_value() {
        assert_eq!(format!("{}", Value::Int(42)), "42");
        assert_eq!(format!("{}", Value::Int(-3)), "-3");
        assert_eq!(format!("{}", Value::Bool(true)), "true");
        assert_eq!(format!("{}", Value::Bool(false)), "false");
    }

    #[test]
    fn short_circuit() {
        // right side has unbound ident; should not be evaluated
        assert_eq!(run("false and x").unwrap(), Value::Bool(false));
        assert_eq!(run("true or x").unwrap(), Value::Bool(true));
    }

    #[test]
    fn keyword_not_identifier() {
        // `let` cannot be used as an identifier
        assert!(parse("let let = 1 in let").is_err());
    }
}
```
