- `pub fn from_roman(s: &str) -> Result<u32, RomanError>`

//! Roman numeral converter.
//!
//! Provides `to_roman` and `from_roman` for the canonical subtractive form
//! over the inclusive range `1..=3999`.

use std::fmt;

/// Error type returned by `to_roman` and `from_roman`.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    /// The integer value is outside the supported range `1..=3999`.
    OutOfRange(u32),
    /// The input string was empty.
    Empty,
    /// The input string contained a character that is not one of
    /// `I`, `V`, `X`, `L`, `C`, `D`, `M`.
    InvalidCharacter(char),
    /// The input string is not the canonical subtractive form for any
    /// integer in `1..=3999` (e.g. `IIII`, `VV`, `IC`).
    NotCanonical(String),
}

impl fmt::Display for RomanError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            RomanError::OutOfRange(n) => {
                write!(f, "value {} is out of range (must be 1..=3999)", n)
            }
            RomanError::Empty => write!(f, "empty roman numeral string"),
            RomanError::InvalidCharacter(c) => {
                write!(f, "invalid roman numeral character: '{}'", c)
            }
            RomanError::NotCanonical(s) => {
                write!(f, "'{}' is not a canonical roman numeral", s)
            }
        }
    }
}

/// Greedy subtractive table, in descending order of value.
const PAIRS: [(u32, &str); 13] = [
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
];

/// Convert an integer in `1..=3999` to its canonical Roman numeral string.
pub fn to_roman(n: u32) -> Result<String, RomanError> {
    if n == 0 || n > 3999 {
        return Err(RomanError::OutOfRange(n));
    }
    let mut out = String::new();
    let mut remaining = n;
    for &(value, symbol) in PAIRS.iter() {
        while remaining >= value {
            out.push_str(symbol);
            remaining -= value;
        }
    }
    Ok(out)
}

/// Numeric value of a single Roman digit, or `None` if the character is not
/// one of the seven allowed uppercase symbols.
fn char_value(c: char) -> Option<u32> {
    match c {
        'I' => Some(1),
        'V' => Some(5),
        'X' => Some(10),
        'L' => Some(50),
        'C' => Some(100),
        'D' => Some(500),
        'M' => Some(1000),
        _ => None,
    }
}

/// Convert a canonical Roman numeral string to its integer value.
///
/// The input is accepted only if it is exactly the string that `to_roman`
/// would produce for some `n` in `1..=3999`. This excludes lowercase input,
/// non-canonical forms such as `IIII` or `VV`, and illegal subtractive
/// pairs such as `IC` or `IL`.
pub fn from_roman(s: &str) -> Result<u32, RomanError> {
    if s.is_empty() {
        return Err(RomanError::Empty);
    }

    // First pass: validate the alphabet.
    for c in s.chars() {
        if char_value(c).is_none() {
            return Err(RomanError::InvalidCharacter(c));
        }
    }

    // Second pass: parse using the standard left-to-right subtractive rule.
    // We use saturating arithmetic so that pathologically long inputs cannot
    // overflow `u32`; any overflow is caught by the canonical round-trip
    // check below (since `to_roman` rejects values above 3999).
    let chars: Vec<char> = s.chars().collect();
    let mut total: u32 = 0;
    let mut i = 0;
    while i < chars.len() {
        let v = char_value(chars[i]).expect("validated above");
        if i + 1 < chars.len() {
            let v_next = char_value(chars[i + 1]).expect("validated above");
            if v < v_next {
                total = total.saturating_add(v_next - v);
                i += 2;
                continue;
            }
        }
        total = total.saturating_add(v);
        i += 1;
    }

    // Canonicality check by round-trip: a string is accepted iff it is
    // *exactly* what `to_roman` would emit for the parsed value. This
    // single check rejects `IIII`, `VV`, `IC`, `IL`, `VX`, etc., and any
    // value outside `1..=3999`.
    let canonical = match to_roman(total) {
        Ok(c) => c,
        Err(_) => return Err(RomanError::NotCanonical(s.to_string())),
    };
    if canonical != s {
        return Err(RomanError::NotCanonical(s.to_string()));
    }

    Ok(total)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn to_roman_known_values() {
        assert_eq!(to_roman(1).unwrap(), "I");
        assert_eq!(to_roman(4).unwrap(), "IV");
        assert_eq!(to_roman(9).unwrap(), "IX");
        assert_eq!(to_roman(40).unwrap(), "XL");
        assert_eq!(to_roman(90).unwrap(), "XC");
        assert_eq!(to_roman(400).unwrap(), "CD");
        assert_eq!(to_roman(900).unwrap(), "CM");
        assert_eq!(to_roman(1994).unwrap(), "MCMXCIV");
        assert_eq!(to_roman(3999).unwrap(), "MMMCMXCIX");
        assert_eq!(to_roman(58).unwrap(), "LVIII");
    }

    #[test]
    fn to_roman_out_of_range() {
        assert!(matches!(to_roman(0), Err(RomanError::OutOfRange(0))));
        assert!(matches!(to_roman(4000), Err(RomanError::OutOfRange(4000))));
        assert!(matches!(
            to_roman(u32::MAX),
            Err(RomanError::OutOfRange(_))
        ));
    }

    #[test]
    fn from_roman_known_values() {
        assert_eq!(from_roman("I").unwrap(), 1);
        assert_eq!(from_roman("IV").unwrap(), 4);
        assert_eq!(from_roman("IX").unwrap(), 9);
        assert_eq!(from_roman("LVIII").unwrap(), 58);
        assert_eq!(from_roman("MCMXCIV").unwrap(), 1994);
        assert_eq!(from_roman("MMMCMXCIX").unwrap(), 3999);
    }

    #[test]
    fn from_roman_empty() {
        assert!(matches!(from_roman(""), Err(RomanError::Empty)));
    }

    #[test]
    fn from_roman_invalid_character() {
        assert!(matches!(
            from_roman("ABC"),
            Err(RomanError::InvalidCharacter('A'))
        ));
        assert!(matches!(
            from_roman("iv"),
            Err(RomanError::InvalidCharacter('i'))
        ));
        assert!(matches!(
            from_roman("X1"),
            Err(RomanError::InvalidCharacter('1'))
        ));
    }

    #[test]
    fn from_roman_not_canonical() {
        for bad in [
            "IIII", "VV", "LL", "DD", "IC", "IL", "XD", "XM", "VX", "VL", "VC",
            "MMMM", "IIV", "IVI", "CMM", "MCMM",
        ] {
            assert!(
                matches!(from_roman(bad), Err(RomanError::NotCanonical(_))),
                "expected NotCanonical for {:?}",
                bad
            );
        }
    }

    #[test]
    fn round_trip_full_range() {
        for n in 1u32..=3999 {
            let s = to_roman(n).expect("to_roman in range");
            let m = from_roman(&s).expect("from_roman canonical");
            assert_eq!(m, n, "round trip failed for {}", n);
        }
    }

    #[test]
    fn display_and_debug() {
        let e = RomanError::OutOfRange(4000);
        let _ = format!("{}", e);
        let _ = format!("{:?}", e);
    }
}
