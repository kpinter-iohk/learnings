```rust
//! Roman numeral converter.
//!
//! Provides conversion between `u32` values in the range `1..=3999` and their
//! canonical Roman numeral string representation, using subtractive notation
//! (`IV`, `IX`, `XL`, `XC`, `CD`, `CM`) and only the seven uppercase symbols
//! `I V X L C D M`.

use std::fmt;

/// Errors produced by [`to_roman`] and [`from_roman`].
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    /// `to_roman` was given a value outside `1..=3999`.
    OutOfRange(u32),
    /// `from_roman` was given the empty string.
    Empty,
    /// `from_roman` encountered a character outside the allowed seven uppercase symbols.
    InvalidCharacter(char),
    /// `from_roman` was given a string that is not the canonical subtractive
    /// form of any value in `1..=3999` (e.g. `IIII`, `VV`, `IC`).
    NotCanonical,
}

impl fmt::Display for RomanError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            RomanError::OutOfRange(n) => {
                write!(f, "value {} is out of range (must be 1..=3999)", n)
            }
            RomanError::Empty => write!(f, "empty input"),
            RomanError::InvalidCharacter(c) => {
                write!(f, "invalid character {:?} (allowed: I V X L C D M)", c)
            }
            RomanError::NotCanonical => {
                write!(f, "input is not a canonical Roman numeral in 1..=3999")
            }
        }
    }
}

/// Ordered list of (value, symbol) pairs used by the greedy encoder.
/// The order is critical: largest first, with subtractive pairs interleaved.
const PAIRS: &[(u32, &str)] = &[
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
];

/// Convert a `u32` in `1..=3999` to its canonical Roman numeral string.
pub fn to_roman(n: u32) -> Result<String, RomanError> {
    if n == 0 || n > 3999 {
        return Err(RomanError::OutOfRange(n));
    }

    let mut out = String::new();
    let mut remaining = n;
    for &(value, sym) in PAIRS {
        while remaining >= value {
            out.push_str(sym);
            remaining -= value;
        }
    }
    debug_assert_eq!(remaining, 0);
    Ok(out)
}

/// Convert a canonical Roman numeral string to its `u32` value.
///
/// Only the canonical subtractive form is accepted; any other spelling
/// (including lowercase, `IIII`, `VV`, `IC`, `XM`, …) is rejected with
/// [`RomanError::NotCanonical`].
pub fn from_roman(s: &str) -> Result<u32, RomanError> {
    if s.is_empty() {
        return Err(RomanError::Empty);
    }

    // Reject any disallowed character up front so the caller gets a precise
    // error pointing at the offending symbol.
    for c in s.chars() {
        if !matches!(c, 'I' | 'V' | 'X' | 'L' | 'C' | 'D' | 'M') {
            return Err(RomanError::InvalidCharacter(c));
        }
    }

    // Standard subtractive parse: if a smaller symbol immediately precedes a
    // larger one, subtract; otherwise add. We use checked arithmetic to avoid
    // any chance of overflow on pathological inputs.
    let bytes = s.as_bytes(); // safe: every char is ASCII after the check above
    let mut total: u32 = 0;
    let mut i = 0;
    while i < bytes.len() {
        let cur = symbol_value(bytes[i]);
        let next = bytes.get(i + 1).map(|&b| symbol_value(b));
        let step = match next {
            Some(nv) if cur < nv => {
                i += 2;
                nv - cur
            }
            _ => {
                i += 1;
                cur
            }
        };
        total = total.checked_add(step).ok_or(RomanError::NotCanonical)?;
        if total > 3999 {
            return Err(RomanError::NotCanonical);
        }
    }

    if total == 0 {
        // Unreachable for non-empty valid-symbol input, but keep the guard for
        // safety: any non-empty Roman string contributes at least 1.
        return Err(RomanError::NotCanonical);
    }

    // Canonicality check: re-encode the parsed value and demand byte-exact
    // equality with the input. This rejects every non-canonical spelling
    // (IIII, VV, IC, XM, IIIII, …) without enumerating them.
    let canonical = to_roman(total).map_err(|_| RomanError::NotCanonical)?;
    if canonical != s {
        return Err(RomanError::NotCanonical);
    }

    Ok(total)
}

#[inline]
fn symbol_value(b: u8) -> u32 {
    match b {
        b'I' => 1,
        b'V' => 5,
        b'X' => 10,
        b'L' => 50,
        b'C' => 100,
        b'D' => 500,
        b'M' => 1000,
        _ => unreachable!("symbol_value called with non-Roman byte"),
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn to_roman_known_values() {
        assert_eq!(to_roman(1).unwrap(), "I");
        assert_eq!(to_roman(2).unwrap(), "II");
        assert_eq!(to_roman(3).unwrap(), "III");
        assert_eq!(to_roman(4).unwrap(), "IV");
        assert_eq!(to_roman(5).unwrap(), "V");
        assert_eq!(to_roman(9).unwrap(), "IX");
        assert_eq!(to_roman(40).unwrap(), "XL");
        assert_eq!(to_roman(49).unwrap(), "XLIX");
        assert_eq!(to_roman(90).unwrap(), "XC");
        assert_eq!(to_roman(400).unwrap(), "CD");
        assert_eq!(to_roman(900).unwrap(), "CM");
        assert_eq!(to_roman(1994).unwrap(), "MCMXCIV");
        assert_eq!(to_roman(3999).unwrap(), "MMMCMXCIX");
    }

    #[test]
    fn to_roman_out_of_range() {
        assert_eq!(to_roman(0), Err(RomanError::OutOfRange(0)));
        assert_eq!(to_roman(4000), Err(RomanError::OutOfRange(4000)));
        assert_eq!(to_roman(u32::MAX), Err(RomanError::OutOfRange(u32::MAX)));
    }

    #[test]
    fn from_roman_known_values() {
        assert_eq!(from_roman("I").unwrap(), 1);
        assert_eq!(from_roman("IV").unwrap(), 4);
        assert_eq!(from_roman("IX").unwrap(), 9);
        assert_eq!(from_roman("XLIX").unwrap(), 49);
        assert_eq!(from_roman("MCMXCIV").unwrap(), 1994);
        assert_eq!(from_roman("MMMCMXCIX").unwrap(), 3999);
    }

    #[test]
    fn from_roman_empty() {
        assert_eq!(from_roman(""), Err(RomanError::Empty));
    }

    #[test]
    fn from_roman_bad_chars() {
        assert!(matches!(
            from_roman("ABC"),
            Err(RomanError::InvalidCharacter('A'))
        ));
        assert!(matches!(
            from_roman("iv"),
            Err(RomanError::InvalidCharacter('i'))
        ));
        assert!(matches!(
            from_roman("XIV "),
            Err(RomanError::InvalidCharacter(' '))
        ));
        assert!(matches!(
            from_roman("X.I"),
            Err(RomanError::InvalidCharacter('.'))
        ));
    }

    #[test]
    fn from_roman_not_canonical() {
        for bad in [
            "IIII", "VV", "LL", "DD", "IC", "IL", "XM", "XD", "VX", "VL", "VC",
            "IIX", "IXIX", "MMMM", "CCCC", "XXXX", "IIIIII",
        ] {
            assert_eq!(
                from_roman(bad),
                Err(RomanError::NotCanonical),
                "expected NotCanonical for {:?}",
                bad
            );
        }
    }

    #[test]
    fn round_trip_full_range() {
        for n in 1u32..=3999 {
            let s = to_roman(n).expect("to_roman in range");
            let back = from_roman(&s).expect("from_roman of canonical form");
            assert_eq!(back, n, "round trip failed for {}: {:?}", n, s);
        }
    }

    #[test]
    fn display_impl_works() {
        let _ = format!("{}", RomanError::OutOfRange(0));
        let _ = format!("{}", RomanError::Empty);
        let _ = format!("{}", RomanError::InvalidCharacter('z'));
        let _ = format!("{}", RomanError::NotCanonical);
    }
}
```
