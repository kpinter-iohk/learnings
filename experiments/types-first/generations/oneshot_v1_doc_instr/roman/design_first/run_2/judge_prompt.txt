//! Roman numeral converter (1..=3999) using canonical subtractive notation.

use std::fmt;

/// Error type returned by [`to_roman`] and [`from_roman`].
///
/// The variants describe each failure mode independently, so callers can
/// distinguish e.g. an out-of-range integer from a malformed string.
#[derive(Debug, PartialEq, Eq, Clone)]
pub enum RomanError {
    /// `to_roman` was called with `0` or with a value greater than `3999`.
    OutOfRange(u32),
    /// `from_roman` was called with an empty string.
    Empty,
    /// `from_roman` encountered a character outside `{I, V, X, L, C, D, M}`.
    /// Lowercase letters and any other byte fall in this category.
    InvalidChar(char),
    /// The input only uses the seven allowed symbols, but it is not the
    /// canonical subtractive form for any integer in `1..=3999`
    /// (e.g. `IIII`, `VV`, `IC`, `MMMM`).
    NotCanonical,
}

impl fmt::Display for RomanError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            RomanError::OutOfRange(n) => {
                write!(f, "value {} is outside the valid range 1..=3999", n)
            }
            RomanError::Empty => write!(f, "input string is empty"),
            RomanError::InvalidChar(c) => write!(
                f,
                "invalid character {:?}: only I V X L C D M are allowed",
                c
            ),
            RomanError::NotCanonical => {
                write!(f, "input is not a canonical Roman numeral in 1..=3999")
            }
        }
    }
}

/// The greedy decomposition table that drives canonical encoding.
/// Ordered from largest value to smallest so that a single left-to-right
/// pass yields the canonical subtractive form.
const TABLE: &[(u32, &str)] = &[
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    (100, "C"),
    (90, "XC"),
    (50, "L"),
    (40, "XL"),
    (10, "X"),
    (9, "IX"),
    (5, "V"),
    (4, "IV"),
    (1, "I"),
);

/// Numeric value of a single allowed Roman symbol.
/// Returns `None` for any other character.
fn symbol_value(c: char) -> Option<u32> {
    match c {
        'I' => Some(1),
        'V' => Some(5),
        'X' => Some(10),
        'L' => Some(50),
        'C' => Some(100),
        'D' => Some(500),
        'M' => Some(1000),
        _ => None,
    }
}

/// Convert an integer in `1..=3999` into its canonical Roman-numeral form.
///
/// Returns `Err(RomanError::OutOfRange(n))` if `n == 0` or `n > 3999`.
pub fn to_roman(n: u32) -> Result<String, RomanError> {
    if n == 0 || n > 3999 {
        return Err(RomanError::OutOfRange(n));
    }
    let mut remaining = n;
    let mut out = String::new();
    for &(value, glyph) in TABLE {
        while remaining >= value {
            out.push_str(glyph);
            remaining -= value;
        }
    }
    Ok(out)
}

/// Parse a canonical Roman numeral string into the integer it represents.
///
/// Only the canonical subtractive form is accepted; all other inputs
/// (empty, foreign characters, non-canonical sequences) are rejected.
pub fn from_roman(s: &str) -> Result<u32, RomanError> {
    if s.is_empty() {
        return Err(RomanError::Empty);
    }

    // First pass: validate every character.
    for c in s.chars() {
        if symbol_value(c).is_none() {
            return Err(RomanError::InvalidChar(c));
        }
    }

    // Second pass: interpret with the standard "smaller-before-larger means
    // subtraction" rule. Even malformed inputs like `IC` will produce *some*
    // total here; the canonicity check below rejects them.
    let chars: Vec<char> = s.chars().collect();
    let mut total: u32 = 0;
    let mut i = 0;
    while i < chars.len() {
        // Safe: validated above.
        let v = symbol_value(chars[i]).unwrap();
        let add = if i + 1 < chars.len() {
            let v_next = symbol_value(chars[i + 1]).unwrap();
            if v < v_next {
                i += 2;
                v_next - v
            } else {
                i += 1;
                v
            }
        } else {
            i += 1;
            v
        };
        total = total.saturating_add(add);
        if total > 3999 {
            return Err(RomanError::NotCanonical);
        }
    }

    if total == 0 {
        // Cannot actually happen for a non-empty validated string,
        // but handled defensively.
        return Err(RomanError::NotCanonical);
    }

    // Canonicity check: re-encode and compare. Because `to_roman` produces
    // exactly one string per integer, any non-canonical spelling of the same
    // value will fail this comparison.
    let canonical = match to_roman(total) {
        Ok(c) => c,
        Err(_) => return Err(RomanError::NotCanonical),
    };
    if canonical != s {
        return Err(RomanError::NotCanonical);
    }

    Ok(total)
}

mod tests {
    use super::*;

    #[test]
    fn known_values() {
        assert_eq!(to_roman(1).unwrap(), "I");
        assert_eq!(to_roman(4).unwrap(), "IV");
        assert_eq!(to_roman(9).unwrap(), "IX");
        assert_eq!(to_roman(40).unwrap(), "XL");
        assert_eq!(to_roman(90).unwrap(), "XC");
        assert_eq!(to_roman(400).unwrap(), "CD");
        assert_eq!(to_roman(900).unwrap(), "CM");
        assert_eq!(to_roman(1994).unwrap(), "MCMXCIV");
        assert_eq!(to_roman(3999).unwrap(), "MMMCMXCIX");
    }

    #[test]
    fn to_roman_out_of_range() {
        assert_eq!(to_roman(0), Err(RomanError::OutOfRange(0)));
        assert_eq!(to_roman(4000), Err(RomanError::OutOfRange(4000)));
    }

    #[test]
    fn from_roman_rejects_garbage() {
        assert_eq!(from_roman(""), Err(RomanError::Empty));
        assert_eq!(from_roman("iv"), Err(RomanError::InvalidChar('i')));
        assert_eq!(from_roman("XYZ"), Err(RomanError::InvalidChar('Y')));
        assert_eq!(from_roman("IIII"), Err(RomanError::NotCanonical));
        assert_eq!(from_roman("VV"), Err(RomanError::NotCanonical));
        assert_eq!(from_roman("IC"), Err(RomanError::NotCanonical));
        assert_eq!(from_roman("MMMM"), Err(RomanError::NotCanonical));
    }

    #[test]
    fn round_trip_full_range() {
        for n in 1..=3999u32 {
            let s = to_roman(n).unwrap();
            assert_eq!(from_roman(&s), Ok(n), "round-trip failed for {}", n);
        }
    }

    #[test]
    fn display_renders_something() {
        let msg = format!("{}", RomanError::OutOfRange(42));
        assert!(msg.contains("42"));
    }
}
